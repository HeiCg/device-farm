const makeHandler = (eventType: 'log' | 'step' | 'status') =>
  (payload: { jobId: string; data: unknown }) => {
    const correlationId = readAls<string>('correlationId') ?? randomUUID();
    const candidate = { type: eventType, correlationId, v: 1 as const, ts: new Date().toISOString(), payload: payload.data };

    }
    jobBroadcaster.emit(payload.jobId, parsed.data as WsEnvelope);
  };
