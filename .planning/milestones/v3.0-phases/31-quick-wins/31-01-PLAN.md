```typescript
const THREADTIME_RE = /^(\d{2}-\d{2} \d{2}:\d{2}:\d{2}\.\d+)\s+(\d+)\s+(\d+)\s+([VDIWEF])\s+([^:]+):\s?(.*)$/;

const CRASH_MARKERS: readonly RegExp[] = [
  /FATAL EXCEPTION/i,
  /Fatal signal \d+/i,
  /beginning of crash/i,
  /AndroidRuntime: FATAL/i,
];

export interface ParsedLogcat {
  level: 'V' | 'D' | 'I' | 'W' | 'E' | 'F';
  tag: string;
  msg: string;
  isCrash: boolean;
}

/**
 * Parse a single Android threadtime-formatted logcat line.
 *
 * Returns null for non-matching lines (no false-positive crash detection on non-logcat
 * lines like user-app stdout or scrcpy output).
 *
 * Crash detection requires 3-of-3:
 *   1. Threadtime regex matches (implicit — null returned otherwise)
 *   2. Level is exactly 'E' (not 'F', not other priorities — CONTEXT-mandated)
 *   3. At least one crash marker (FATAL EXCEPTION / Fatal signal N / beginning of crash /
 *      AndroidRuntime: FATAL) matches the raw input line
 */
export function parseLogcatLine(line: string): ParsedLogcat | null {
  const m = THREADTIME_RE.exec(line);
  if (!m) return null;
  const level = m[4] as ParsedLogcat['level'];
  const tag = m[5].trim();
  const msg = m[6];
  const isCrash = level === 'E' && CRASH_MARKERS.some((re) => re.test(line));
  return { level, tag, msg, isCrash };
}
```

3. Modify the `makeHandler('log')` body. The current code (per RESEARCH.md Existing 1 pattern) creates a `candidate` envelope and broadcasts it. ADD parser enrichment + crash gate inside the log handler:

```typescript
const handleLog = (payload: { jobId: string; data: unknown }) => {
  const correlationId = readCorrelationIdFromAls() ?? randomUUID();
  const ts = new Date().toISOString();

  // Extract raw line for parsing (payload.data may be string or {line: string})
  const rawLine: string =
    typeof payload.data === 'string'
      ? payload.data
      : (payload.data as { line?: string })?.line ?? '';

  const parsed = parseLogcatLine(rawLine);

  // Enriched payload preserves existing shape; adds `parsed` field only when parser matched
  const enrichedPayload =
    parsed !== null
      ? typeof payload.data === 'string'
        ? { line: rawLine, parsed }
        : { ...(payload.data as object), parsed }
      : payload.data;

  const candidate = {
    type: 'log' as const,
    correlationId,
    v: 1 as const,
    ts,
    payload: enrichedPayload,
  };

  const validated = wsEnvelopeSchema.safeParse(candidate);
  if (!validated.success) {
    logger.warn({ jobId: payload.jobId, err: validated.error }, 'ws.frame.dropped');
    return;
  }
  broadcaster.emit(payload.jobId, validated.data);

  // First-crash gate: fire-and-forget DB + emit crash-detected separately
  if (parsed?.isCrash && !firstCrashSeen.has(payload.jobId)) {
    firstCrashSeen.add(payload.jobId);

    // Fire-and-forget DB update (Phase 23 saga pattern — no await, log on error)
    db.update(jobs)
      .set({ failureClass: 'crash' })
      .where(eq(jobs.id, payload.jobId))
      .catch((err) => {
        logger.error({ err, jobId: payload.jobId }, 'failed to set failure_class=crash');
      });

    // Emit separate crash-detected envelope to the broadcaster
    const crashEnvelope = wsEnvelopeSchema.safeParse({
      type: 'crash-detected',
      correlationId,
      v: 1 as const,
      ts,
      payload: { firstLine: rawLine },
    });
    if (crashEnvelope.success) {
      broadcaster.emit(payload.jobId, crashEnvelope.data);
    }
  }
};
```
