1. Create `server/streaming/internal/flush-queue.ts`:

```typescript
import type { WsEnvelope } from './ws-schemas.js';

const FLUSH_INTERVAL_MS = 150;
const BUFFER_CAP = 64;

/**
 * Minimal WebSocket interface that FlushQueue depends on.
 * Compatible with `ws` library WebSocket and `@fastify/websocket` socket.
 */
export interface SocketLike {
  send(data: string): void;
  readyState: number; // 1 = OPEN
}

/**
 * Per-WS-connection batcher.
 *
 * In batchMode (DEFAULT): coalesces pushed envelopes into a single outer batch
 * envelope `{type:'batch', items: WsEnvelope[]}` flushed on a 150ms timer OR
 * when the buffer reaches 64 entries (whichever fires first).
 *
 * In !batchMode (?nobatch=1 opt-out): sends each envelope unbatched immediately.
 *
 * `close()` clears the timer and drains pending entries (no lost tail).
 *
 * The outer batch envelope is intentionally NOT validated via wsEnvelopeSchema —
 * it's a transport-level wrapper; the INNER items are still strict wsEnvelopeSchema
 * shapes (validated by the broadcaster before they reach push()).
 */
export class FlushQueue {
  private buf: WsEnvelope[] = [];
  private timer: ReturnType<typeof setInterval> | null;

  constructor(
    private readonly socket: SocketLike,
    private readonly batchMode: boolean,
  ) {
    this.timer = batchMode ? setInterval(() => this.flush(), FLUSH_INTERVAL_MS) : null;
  }

  push(env: WsEnvelope): void {
    if (!this.batchMode) {
      this.sendOne(env);
      return;
    }
    this.buf.push(env);
    if (this.buf.length >= BUFFER_CAP) this.flush();
  }

  flush(): void {
    if (this.buf.length === 0) return;
    const items = this.buf;
    this.buf = [];
    this.sendRaw({ type: 'batch', items });
  }

  close(): void {
    if (this.timer) {
      clearInterval(this.timer);
      this.timer = null;
    }
    this.flush();
  }

  private sendOne(env: WsEnvelope): void {
    if (this.socket.readyState === 1) {
      this.socket.send(JSON.stringify(env));
    }
  }

  private sendRaw(payload: unknown): void {
    if (this.socket.readyState === 1) {
      this.socket.send(JSON.stringify(payload));
    }
  }
}
```

```go
// UnwrapBatch decodes the outer transport envelope.
//
// If the outer JSON has `{"type":"batch","items":[...]}` shape, returns each inner
// item as a raw JSON message. Otherwise returns []json.RawMessage{raw} (passthrough)
// so the caller's existing per-frame dispatch logic continues to work.
//
// This is the SC2 Go-side change required by Phase 31: the server now defaults to
// batch envelopes; legacy unbatched frames still pass through unchanged.
func UnwrapBatch(raw []byte) ([]json.RawMessage, error) {
	var peek struct {
		Type  string            `json:"type"`
		Items []json.RawMessage `json:"items"`
	}
	if err := json.Unmarshal(raw, &peek); err != nil {
		return nil, err
	}
	if peek.Type == "batch" {
		if peek.Items == nil {
			return []json.RawMessage{}, nil
		}
		return peek.Items, nil
	}
	return []json.RawMessage{raw}, nil
}
```
