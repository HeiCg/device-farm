     sessionsModule.openSockets.set(sessionId, socket);

     let alive = true;
     const heartbeat = setInterval(() => {
       if (!alive) { socket.terminate(); clearInterval(heartbeat); return; }
       alive = false;
       socket.ping();
     }, 30_000);
     socket.on('pong', () => { alive = true; });

     // Build ActionContext
     const ctx = buildActionContext(fastify, (socket as any).context);

     socket.send(JSON.stringify({ type:'event', kind:'connected', data:{ heartbeatIntervalMs: 30000 }}));

     socket.on('message', async (raw) => {
       const start = Date.now();
       let json;
       try { json = JSON.parse(raw.toString()); }
       catch { logger.warn({ sessionId }, 'WS frame: invalid JSON, dropped'); return; }

       const parsed = clientEnvelope.safeParse(json);
       if (!parsed.success) {
         socket.send(JSON.stringify({ type:'error', forMsgId: json?.id, code:'invalid_envelope', message: parsed.error.issues[0]?.message ?? 'parse failed' }));
         socket.close(1003); return;
       }

       // ALS restore for emit envelopes downstream
       await als.run({ actor: ctx.session.actor, correlationId: randomUUID() }, async () => {
         // Rate-limit stub call — real body in 34-04
         if (!rateLimitOk(sessionId, Date.now())) {
           socket.send(JSON.stringify({ type:'error', forMsgId: parsed.data.id, code:'rate_limited', message:'30 actions / 10s' }));
           return;
         }

         try {
           const result = await dispatch(parsed.data, ctx);
           socket.send(JSON.stringify({ type:'ack', forMsgId: parsed.data.id, durationMs: Date.now() - start, result: result ?? undefined }));
         } catch (err: any) {
           const code = err instanceof ResolverError ? 'resolver_failed' : 'device_error';
           socket.send(JSON.stringify({ type:'error', forMsgId: parsed.data.id, code, message: String(err?.message ?? err) }));
         }
       });
     });

     socket.on('close', () => {
       clearInterval(heartbeat);
       sessionsModule.openSockets.delete(sessionId);
     });
   }
