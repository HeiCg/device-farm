```typescript
export function createRateLimiter(opts: { windowMs?: number; max?: number } = {}) {
  const windowMs = opts.windowMs ?? 10_000;
  const max = opts.max ?? 30;
  const windows = new Map<string, number[]>();

  function check(sessionId: string, now: number): boolean {
    let win = windows.get(sessionId);
    if (!win) { win = []; windows.set(sessionId, win); }
    const cutoff = now - windowMs;
    while (win.length && win[0] < cutoff) win.shift();
    if (win.length >= max) return false;
    win.push(now);
    return true;
  }

  function clear(sessionId: string) { windows.delete(sessionId); }
  function size() { return windows.size; }
  return { check, clear, size };
}
```

   ```typescript
   import { and, eq, lt, sql } from 'drizzle-orm';
   import { QUEUE_NAMES } from '../../queue/names.js';
   import { sessions } from '../../db/schema.js';
   import type { FastifyInstance } from 'fastify';
   import type pino from 'pino';
   import type { SessionsModule } from './module.js';

   export interface SessionSweeperDeps {
     fastify: FastifyInstance;
     sessionsModule: SessionsModule;
     logger: pino.Logger;
   }

   export interface SweeperHandle {
     intervalHandle: NodeJS.Timeout | null;
     strategy: 'cron-30s' | 'cron-1m-plus-interval';
   }

   export async function sweepExpiredSessions(deps: SessionSweeperDeps): Promise<number> {
     const { fastify, sessionsModule, logger } = deps;
     const expired = await fastify.db
       .select()
       .from(sessions)
       .where(and(eq(sessions.status, 'active'), lt(sessions.leaseUntil, sql`now()`)));

     for (const session of expired) {
       const socket = sessionsModule.openSockets.get(session.id);
       if (socket) {
         try {
           socket.send(JSON.stringify({ type: 'event', kind: 'lease-expiring', data: { sessionId: session.id }}));
           // Close after small delay handled by next tick of WS loop; alternatively close here:
           socket.close(1000);
         } catch (err) { logger.warn({ err, sessionId: session.id }, 'sweep: failed to broadcast lease-expiring'); }
       }
       try {
         await fastify.pool.releaseDevice(session.deviceId);
       } catch (err) {
         logger.error({ err, sessionId: session.id, deviceId: session.deviceId }, 'sweep: pool.releaseDevice failed');
       }
       await fastify.db.update(sessions)
         .set({ status: 'expired', releasedAt: new Date() })
         .where(eq(sessions.id, session.id));
       await sessionsModule.emit.expired({
         sessionId: session.id,
         deviceId: session.deviceId,
         leaseUntil: session.leaseUntil.toISOString(),
       });
       sessionsModule.rateLimiter.clear(session.id);
     }

     return expired.length;
   }

   export async function registerSessionSweeper(deps: SessionSweeperDeps): Promise<SweeperHandle> {
     const { fastify, logger } = deps;
     const handle: SweeperHandle = { intervalHandle: null, strategy: 'cron-30s' };

     try {
       await fastify.boss.schedule(QUEUE_NAMES.SESSION_SWEEP, '*/30 * * * * *', {}, { tz: 'UTC' });
       handle.strategy = 'cron-30s';
       logger.info({ schedule: 'cron-30s' }, 'session sweeper: pg-boss 6-field cron accepted');
     } catch (err) {
       logger.warn({ err: String(err) }, 'session sweeper: 6-field cron rejected, falling back to cron-1m + 30s interval');
       await fastify.boss.schedule(QUEUE_NAMES.SESSION_SWEEP, '* * * * *', {}, { tz: 'UTC' });
       handle.intervalHandle = setInterval(() => {
         sweepExpiredSessions(deps).catch((e) => logger.error({ err: e }, 'sweep interval tick failed'));
       }, 30_000);
       handle.strategy = 'cron-1m-plus-interval';
     }

     await fastify.boss.work(QUEUE_NAMES.SESSION_SWEEP, { teamSize: 1 }, async () => {
       try { await sweepExpiredSessions(deps); }
       catch (err) { logger.error({ err }, 'sweep worker tick failed'); }
     });

     return handle;
   }
   ```
