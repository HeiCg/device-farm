1. Create `cli/internal/session/persist.go`:
   ```go
   package session

   import (
     "encoding/json"
     "errors"
     "os"
     "path/filepath"
     "time"
   )

   type Ref struct {
     SessionID  string    `json:"sessionId"`
     WSUrl      string    `json:"wsUrl"`
     Token      string    `json:"token"`
     DeviceID   string    `json:"deviceId"`
     DeviceName string    `json:"deviceName"`
     LeaseUntil time.Time `json:"leaseUntil"`
     Platform   string    `json:"platform"`
   }

   func persistPath() (string, error) {
     home, err := os.UserHomeDir()
     if err != nil { return "", err }
     return filepath.Join(home, ".device-farm", "session.json"), nil
   }

   func Load() (*Ref, error) {
     p, err := persistPath()
     if err != nil { return nil, err }
     b, err := os.ReadFile(p)
     if err != nil { return nil, err }  // includes os.ErrNotExist
     var r Ref
     if err := json.Unmarshal(b, &r); err != nil { return nil, err }
     return &r, nil
   }

   func Save(ref *Ref) error {
     p, err := persistPath()
     if err != nil { return err }
     if err := os.MkdirAll(filepath.Dir(p), 0o700); err != nil { return err }
     b, err := json.MarshalIndent(ref, "", "  ")
     if err != nil { return err }
     return os.WriteFile(p, b, 0o600)
   }

   func Clear() error {
     p, err := persistPath()
     if err != nil { return err }
     if err := os.Remove(p); err != nil && !errors.Is(err, os.ErrNotExist) { return err }
     return nil
   }
   ```

2. Create `cli/internal/session/client.go`:
   ```go
   package session

   import (
     "bytes"
     "context"
     "encoding/json"
     "errors"
     "fmt"
     "io"
     "net/http"
     "time"

     "github.com/google/uuid"
     "github.com/gorilla/websocket"
   )

   var actionTimeout = 30 * time.Second  // test override hook

   type LeaseRequest struct {
     Platform     string         `json:"platform"`
     TTLSeconds   int            `json:"ttlSeconds,omitempty"`
     DeviceQuery  map[string]any `json:"deviceQuery,omitempty"`
     Metadata     map[string]any `json:"metadata,omitempty"`
   }

   func Lease(serverURL, token string, body LeaseRequest) (*Ref, error) {
     buf, _ := json.Marshal(body)
     req, _ := http.NewRequest(http.MethodPost, serverURL+"/api/sessions", bytes.NewReader(buf))
     req.Header.Set("Authorization", "Bearer "+token)
     req.Header.Set("Content-Type", "application/json")
     resp, err := http.DefaultClient.Do(req)
     if err != nil { return nil, err }
     defer resp.Body.Close()
     if resp.StatusCode != http.StatusOK {
       b, _ := io.ReadAll(resp.Body)
       return nil, fmt.Errorf("lease failed %d: %s", resp.StatusCode, string(b))
     }
     var ref Ref
     if err := json.NewDecoder(resp.Body).Decode(&ref); err != nil { return nil, err }
     ref.Token = token
     return &ref, nil
   }

   func Release(serverURL, token, sessionID string) error {
     req, _ := http.NewRequest(http.MethodDelete, serverURL+"/api/sessions/"+sessionID, nil)
     req.Header.Set("Authorization", "Bearer "+token)
     resp, err := http.DefaultClient.Do(req)
     if err != nil { return err }
     defer resp.Body.Close()
     if resp.StatusCode >= 400 { return fmt.Errorf("release failed %d", resp.StatusCode) }
     return nil
   }

   func SendAction(wsURL, token string, envelope map[string]any) (map[string]any, error) {
     if _, ok := envelope["id"]; !ok { envelope["id"] = uuid.NewString() }
     wantID := envelope["id"].(string)

     header := http.Header{}
     header.Set("Authorization", "Bearer "+token)
     conn, _, err := websocket.DefaultDialer.Dial(wsURL+"?token="+token, header)
     if err != nil { return nil, err }
     defer conn.Close()

     if err := conn.WriteJSON(envelope); err != nil { return nil, err }

     ctx, cancel := context.WithTimeout(context.Background(), actionTimeout)
     defer cancel()
     done := make(chan struct{}, 1)
     var result map[string]any
     var readErr error

     go func() {
       for {
         _, raw, err := conn.ReadMessage()
         if err != nil { readErr = err; done <- struct{}{}; return }
         var frame map[string]any
         if err := json.Unmarshal(raw, &frame); err != nil { continue }
         if frame["forMsgId"] != wantID { continue }
         if frame["type"] == "ack" { result = frame; done <- struct{}{}; return }
         if frame["type"] == "error" {
           readErr = fmt.Errorf("%v: %v", frame["code"], frame["message"])
           done <- struct{}{}; return
         }
       }
     }()

     select {
     case <-done:
       if readErr != nil { return nil, readErr }
       return result, nil
     case <-ctx.Done():
       return nil, errors.New("action timeout")
     }
   }
   ```
