   ```typescript
   import { XMLParser } from 'fast-xml-parser';
   import type { ResolveTargetRequest, ResolveTargetResult, TargetResolver } from './types.js';

   const STOPWORDS = new Set(['the','a','an','and','or','on','in','to','for','button','field','text','icon']);

   function tokenize(s: string): string[] {
     return s.toLowerCase().split(/\W+/).filter(t => t.length > 0 && !STOPWORDS.has(t));
   }

   function parseBounds(s: string | undefined): { x1: number; y1: number; x2: number; y2: number } | null {
     if (!s) return null;
     const m = s.match(/^\[(\d+),(\d+)\]\[(\d+),(\d+)\]$/);
     return m ? { x1: +m[1], y1: +m[2], x2: +m[3], y2: +m[4] } : null;
   }

   function* walkAndroid(node: any): Generator<any> {
     if (!node) return;
     if (Array.isArray(node)) { for (const n of node) yield* walkAndroid(n); return; }
     if (node['@_bounds'] !== undefined) yield node;
     for (const k of Object.keys(node)) {
       if (k.startsWith('@_')) continue;
       yield* walkAndroid(node[k]);
     }
   }

   function scoreAndroidNode(node: any, tokens: string[]): number {
     const candidates = [
       (node['@_text'] ?? '').toLowerCase(),
       (node['@_content-desc'] ?? '').toLowerCase(),
       (node['@_resource-id'] ?? '').toLowerCase(),
     ];
     let score = 0;
     for (const tok of tokens) {
       for (const c of candidates) {
         if (c.includes(tok)) score += 1;
       }
     }
     if ((node['@_clickable'] ?? 'false') !== 'true') score -= 2;
     // Class boost: Button > TextView
     if ((node['@_class'] ?? '').includes('Button')) score += 0.5;
     return score;
   }

   export class MaestroAiResolver implements TargetResolver {
     async resolve(req: ResolveTargetRequest): Promise<ResolveTargetResult> {
       const t0 = Date.now();
       const tokens = tokenize(req.target);
       if (tokens.length === 0) {
         return { x: 0, y: 0, confidence: 0, backend: 'maestro-ai', cached: false, latencyMs: Date.now() - t0 };
       }

       let best: { node: any; score: number } | null = null;
       if (req.platform === 'android') {
         const parser = new XMLParser({ ignoreAttributes: false, attributeNamePrefix: '@_' });
         const tree = parser.parse(req.hierarchy);
         for (const node of walkAndroid(tree)) {
           const s = scoreAndroidNode(node, tokens);
           if (s > 0 && (!best || s > best.score)) best = { node, score: s };
         }
       } else {
         // iOS: hierarchy is JSON; structure varies — walk similarly
         try {
           const tree = JSON.parse(req.hierarchy);
           // Same walk but use {label, value, identifier} fields and {frame: {x, y, width, height}}
           // ... (implementation analogous to Android walker)
         } catch {
           return { x: 0, y: 0, confidence: 0.3, backend: 'maestro-ai', cached: false, latencyMs: Date.now() - t0 };
         }
       }

       if (!best || best.score < 1) {
         return { x: 0, y: 0, confidence: 0.3, backend: 'maestro-ai', cached: false, latencyMs: Date.now() - t0 };
       }

       const bounds = parseBounds(best.node['@_bounds']);
       if (!bounds) {
         return { x: 0, y: 0, confidence: 0.3, backend: 'maestro-ai', cached: false, latencyMs: Date.now() - t0 };
       }

       const x = Math.floor((bounds.x1 + bounds.x2) / 2);
       const y = Math.floor((bounds.y1 + bounds.y2) / 2);
       const confidence = Math.min(0.95, 0.5 + best.score * 0.15);
       return { x, y, confidence, backend: 'maestro-ai', cached: false, latencyMs: Date.now() - t0 };
     }
   }
   ```

1. Create `server/sessions/internal/resolver/claude-vision.ts`:

   ```typescript
   import Anthropic from '@anthropic-ai/sdk';
   import { LRUCache } from 'lru-cache';
   import { createHash } from 'node:crypto';
   import type { ResolveTargetRequest, ResolveTargetResult, TargetResolver } from './types.js';
   import { ResolverError } from './types.js';

   const cache = new LRUCache<string, ResolveTargetResult>({ max: 100, ttl: 5 * 60 * 1000 });

   export interface ClaudeVisionResolverDeps {
     apiKey?: string;
     model?: string;
     anthropicClient?: Anthropic;  // injection point for tests
   }

   export class ClaudeVisionResolver implements TargetResolver {
     private client: Anthropic;
     private model: string;

     constructor(deps: ClaudeVisionResolverDeps = {}) {
       const apiKey = deps.apiKey ?? process.env.ANTHROPIC_API_KEY;
       if (!apiKey && !deps.anthropicClient) {
         throw new Error('ANTHROPIC_API_KEY not set');
       }
       this.client = deps.anthropicClient ?? new Anthropic({ apiKey });
       this.model = deps.model ?? process.env.SESSION_RESOLVER_MODEL ?? 'claude-sonnet-4-5';
     }

     async resolve(req: ResolveTargetRequest): Promise<ResolveTargetResult> {
       const t0 = Date.now();
       const sha = createHash('sha256').update(req.screenshot).digest('hex');
       const key = `${sha}:${req.target}`;
       const cached = cache.get(key);
       if (cached) return { ...cached, cached: true, latencyMs: Date.now() - t0 };

       const response = await this.client.messages.create({
         model: this.model,
         max_tokens: 256,
         messages: [{
           role: 'user',
           content: [
             { type: 'image', source: { type: 'base64', media_type: 'image/png', data: req.screenshot.toString('base64') } },
             { type: 'text', text: `You are a UI element locator for a ${req.platform} device. Given a mobile screenshot (${req.screenWidth}x${req.screenHeight}px, origin top-left) and a target description, return ONLY JSON: {"x": <0-1 normalized>, "y": <0-1 normalized>, "confidence": <0-1>}. Coordinates are the center of the element. If not found, return {"x": null, "y": null, "confidence": 0}.\n\nHierarchy hint (use to disambiguate when image is ambiguous):\n${req.hierarchy.slice(0, 4000)}\n\nTarget: ${req.target}` },
           ],
         }],
       });

       const textBlock = response.content.find((c: any) => c.type === 'text');
       if (!textBlock || textBlock.type !== 'text') {
         throw new ResolverError('Anthropic returned non-text response', { confidence: 0 });
       }
       let parsed: any;
       try { parsed = JSON.parse(textBlock.text); }
       catch { throw new ResolverError(`malformed JSON: ${textBlock.text}`, { confidence: 0 }); }

       const result: ResolveTargetResult = {
         x: parsed.x == null ? 0 : Math.floor(parsed.x * req.screenWidth),
         y: parsed.y == null ? 0 : Math.floor(parsed.y * req.screenHeight),
         confidence: typeof parsed.confidence === 'number' ? parsed.confidence : 0,
         backend: 'claude-vision',
         cached: false,
         latencyMs: Date.now() - t0,
       };
       cache.set(key, result);
       return result;
     }
   }
   ```
