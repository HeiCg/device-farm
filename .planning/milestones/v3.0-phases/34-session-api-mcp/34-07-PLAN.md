1. Create `web/src/lib/sessions/ws.ts`:
   ```typescript
   import { writable, type Readable } from 'svelte/store';

   export interface AckResult { durationMs: number; result?: unknown; }

   export function isLoopback(hostname: string): boolean {
     return hostname === '127.0.0.1' || hostname === 'localhost';
   }

   // TLS-first scheme guard — RESEARCH §Open Q #8.
   // Production deployments terminate TLS at the server or upstream proxy; plaintext
   // is permitted ONLY for loopback dev hosts. Any other plaintext is force-upgraded
   // to wss: to prevent accidental cleartext token transmission.
   export function deriveWsScheme(loc: { protocol: string; hostname: string }): 'wss:' | 'ws:' {
     if (loc.protocol === 'https:') return 'wss:';
     if (isLoopback(loc.hostname)) return 'ws:';
     return 'wss:';
   }

   export function createSessionWs(wsUrl: string, token: string) {
     const finalUrl = `${wsUrl}?token=${encodeURIComponent(token)}`;
     const socket = new WebSocket(finalUrl);
     const frames = writable<any[]>([]);
     const pending = new Map<string, { resolve: (a: AckResult) => void; reject: (e: any) => void }>();

     socket.addEventListener('message', (ev) => {
       try {
         const frame = JSON.parse(typeof ev.data === 'string' ? ev.data : ev.data.toString());
         frames.update((arr) => [...arr, { ...frame, _at: Date.now() }]);
         if (frame.type === 'ack' || frame.type === 'pong') {
           const r = pending.get(frame.forMsgId);
           if (r) { pending.delete(frame.forMsgId); r.resolve({ durationMs: frame.durationMs ?? 0, result: frame.result }); }
         } else if (frame.type === 'error' && frame.forMsgId) {
           const r = pending.get(frame.forMsgId);
           if (r) { pending.delete(frame.forMsgId); r.reject(new Error(`${frame.code}: ${frame.message}`)); }
         }
       } catch { /* drop malformed */ }
     });

     async function send(partial: any): Promise<AckResult> {
       const id = partial.id ?? crypto.randomUUID();
       const envelope = { ...partial, id };
       await new Promise<void>((r) => {
         if (socket.readyState === socket.OPEN) r();
         else socket.addEventListener('open', () => r(), { once: true });
       });
       return new Promise<AckResult>((resolve, reject) => {
         pending.set(id, { resolve, reject });
         socket.send(JSON.stringify(envelope));
         setTimeout(() => {
           if (pending.delete(id)) reject(new Error(`timeout id=${id}`));
         }, 30_000);
       });
     }

     return { frames: { subscribe: frames.subscribe } as Readable<any[]>, send, close: () => socket.close(1000) };
   }
   ```
