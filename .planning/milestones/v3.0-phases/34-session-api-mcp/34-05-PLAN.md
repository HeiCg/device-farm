   ```typescript
   import WebSocket from 'ws';
   import { randomUUID } from 'node:crypto';

   interface PendingResolver { resolve(value: any): void; reject(err: any): void; }
   interface WsEntry { socket: WebSocket; pending: Map<string, PendingResolver>; wsUrl: string; }

   export class DeviceFarmClient {
     private wsBySession = new Map<string, WsEntry>();
     private wsUrlBySession = new Map<string, string>();  // populated ONLY by lease()
     constructor(private opts: { baseUrl: string; token: string }) {}

     async lease(body: { platform: 'android'|'ios'; deviceQuery?: any; ttlSeconds?: number; metadata?: any }) {
       const res = await fetch(`${this.opts.baseUrl}/api/sessions`, {
         method: 'POST', headers: { 'Authorization': `Bearer ${this.opts.token}`, 'Content-Type': 'application/json' },
         body: JSON.stringify(body),
       });
       if (!res.ok) throw new Error(`lease failed ${res.status}: ${await res.text()}`);
       const ref = await res.json();
       this.wsUrlBySession.set(ref.sessionId, ref.wsUrl);  // CACHE FROM LEASE RESPONSE
       return ref;
     }

     async release(sessionId: string) {
       const res = await fetch(`${this.opts.baseUrl}/api/sessions/${sessionId}`, {
         method: 'DELETE', headers: { 'Authorization': `Bearer ${this.opts.token}` },
       });
       this.wsUrlBySession.delete(sessionId);
       const existing = this.wsBySession.get(sessionId);
       if (existing) { existing.socket.close(1000); this.wsBySession.delete(sessionId); }
       if (!res.ok) throw new Error(`release failed ${res.status}`);
       return res.json();
     }

     async listSessions() { /* GET /api/sessions — read-only; does NOT populate wsUrlBySession */ }
     async listDevices() { /* GET /api/devices */ }

     async sendAction(sessionId: string, partial: any): Promise<unknown> {
       const id = partial.id ?? randomUUID();
       const envelope = { ...partial, id };
       const entry = await this.getOrOpenWs(sessionId);
       return new Promise((resolve, reject) => {
         entry.pending.set(id, { resolve, reject });
         entry.socket.send(JSON.stringify(envelope));
         const timeout = setTimeout(() => {
           if (entry.pending.delete(id)) reject(new Error(`action timeout (id=${id})`));
         }, 30_000);
         // (timeout cleared on resolve via wrapper — omitted for brevity)
       });
     }

     private async getOrOpenWs(sessionId: string): Promise<WsEntry> {
       const existing = this.wsBySession.get(sessionId);
       if (existing && existing.socket.readyState === existing.socket.OPEN) return existing;

       // CACHE LOOKUP — never GET /api/sessions to discover the URL.
       const wsUrl = this.wsUrlBySession.get(sessionId);
       if (!wsUrl) throw new Error('session not leased by this client');

       const socket = new WebSocket(wsUrl);
       const pending = new Map<string, PendingResolver>();
       const entry: WsEntry = { socket, pending, wsUrl };
       socket.on('message', (raw) => {
         try {
           const frame = JSON.parse(raw.toString());
           if (frame.type === 'ack' || frame.type === 'pong') {
             const r = pending.get(frame.forMsgId);
             if (r) { pending.delete(frame.forMsgId); r.resolve(frame.result ?? null); }
           } else if (frame.type === 'error' && frame.forMsgId) {
             const r = pending.get(frame.forMsgId);
             if (r) { pending.delete(frame.forMsgId); r.reject(new Error(`${frame.code}: ${frame.message}`)); }
           }
         } catch { /* malformed frame from server — ignore */ }
       });
       await new Promise<void>((res, rej) => { socket.once('open', () => res()); socket.once('error', rej); });
       this.wsBySession.set(sessionId, entry);
       return entry;
     }

     async closeAll() { for (const e of this.wsBySession.values()) e.socket.close(1000); this.wsBySession.clear(); this.wsUrlBySession.clear(); }
   }
   ```
