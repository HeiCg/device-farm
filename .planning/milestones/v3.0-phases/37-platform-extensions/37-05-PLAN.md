```typescript
import type { PreflightModule } from '../../preflight/index.js';
import type { AnalysisModule } from '../../analysis/index.js';

export interface WebhookPayloadV3 {
  // ... existing v3 fields (preserve exactly)

  // Phase 37 additions — all optional, additive per SPEC-08.
  preflight?: {
    pass: boolean;
    rulesVersion: string;
    blockers: Array<{ rule_id: string; severity: 'blocker'; message: string }>;
    warnings: Array<{ rule_id: string; severity: 'warning'; message: string }>;
  } | null;
  analysis?: {
    analysisId: string;
    candidateScreensCount: number;
    deepLinks: Array<{ scheme: string; is_oauth_callback: boolean }>;
  } | null;
  parallelDeploy?: {
    deviceCount: number;
    devices: Array<{ deviceId: string; status: 'passed' | 'failed'; durationMs?: number; error?: string }>;
  } | null;
}

export async function buildWebhookPayload(deps: { /* existing */ preflightModule?: PreflightModule; analysisModule?: AnalysisModule }, job: Job): Promise<WebhookPayloadV3> {
  const base = await buildBasePayload(deps, job); // existing builder

  // Track A: analysis
  let analysis: WebhookPayloadV3['analysis'] = null;
  if (deps.analysisModule) {
    const row = await deps.analysisModule.repo.getByJobId(job.id);
    if (row) {
      analysis = {
        analysisId: row.id,
        candidateScreensCount: row.payload.stats.candidate_screens,
        deepLinks: row.payload.deep_link_entries.map(d => ({ scheme: d.scheme, is_oauth_callback: d.is_oauth_callback })),
      };
    }
  }

  // Track B: preflight — look up by job metadata.preflightRunId if associated
  let preflight: WebhookPayloadV3['preflight'] = null;
  const preflightRunId = (job.metadata as { preflightRunId?: string } | null)?.preflightRunId;
  if (preflightRunId && deps.preflightModule) {
    const row = await deps.preflightModule.repo.getById(preflightRunId);
    if (row) {
      preflight = {
        pass: row.status !== 'blocked',
        rulesVersion: row.rulesVersion,
        blockers: row.blockers,
        warnings: row.warnings,
      };
    }
  }

  // Track D: parallel deploy — read from job.metadata.parallelDeploy if present
  const parallelDeploy = (job.metadata as { parallelDeploy?: WebhookPayloadV3['parallelDeploy'] } | null)?.parallelDeploy ?? null;

  return { ...base, analysis, preflight, parallelDeploy };
}
```
