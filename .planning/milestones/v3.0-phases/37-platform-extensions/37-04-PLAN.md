```typescript
import type pino from 'pino';
import type { SessionsModule, NormalizedTouch, KeyEvent } from '../../sessions/index.js';

export type BroadcastAction =
  | { type: 'tap'; touch: NormalizedTouch }
  | { type: 'key'; event: KeyEvent }
  | { type: 'text'; text: string };

export interface BroadcastResult {
  sessionId: string;
  ok: boolean;
  error?: string;
}

export interface InputBroadcaster {
  broadcast(sessionIds: string[], action: BroadcastAction): Promise<BroadcastResult[]>;
}

export function createInputBroadcaster(deps: { sessionsModule: SessionsModule; logger: pino.Logger }): InputBroadcaster {
  return {
    async broadcast(sessionIds, action) {
      // Snapshot sessionIds; validate existence in dispatch (Pitfall 8 — race-safe).
      // No mutex around the snapshot — racy disposes surface as per-session NOT_FOUND
      // in the result array, mirroring kittyfarm's "error: <name>" + continue pattern.
      const results = await Promise.allSettled(
        sessionIds.map((id) => deps.sessionsModule.dispatch(id, action)),
      );
      return results.map((r, i): BroadcastResult => r.status === 'fulfilled'
        ? { sessionId: sessionIds[i], ok: true }
        : { sessionId: sessionIds[i], ok: false, error: r.reason instanceof Error ? r.reason.message : String(r.reason) },
      );
    },
  };
}
```

  ]),

export function registerSessionBroadcastRoute(app: FastifyInstance, deps: { broadcaster: InputBroadcaster }) {
  app.post('/api/sessions/broadcast', {
    schema: { body: broadcastBodySchema, response: { 200: z.object({ results: z.array(/* ... */) }) } },
  }, async (req, reply) => {
    const { sessionIds, action } = req.body;
    const results = await deps.broadcaster.broadcast(sessionIds, action);
    return { results };

**server/pool/pool-manager.ts:** Implement `allocateMany(platform, count)` on top of the existing single-device `allocate`:
```typescript
import type { AllocatedDevice, Platform } from './types.js';

// On the PoolManager class (or factory return object):
async allocateMany(platform: Platform, count: number): Promise<AllocatedDevice[]> {
  // Use Promise.allSettled so partial failures (e.g. only N-1 devices healthy)
  // surface to the caller without rolling back successful allocations. Caller
  // (runParallelDeploy / route layer) decides whether to release the partial
  // set or proceed. Pitfall 9 cap is enforced at the route layer BEFORE this
  // is called; this method is defensive against races.
  const settled = await Promise.allSettled(
    Array.from({ length: count }, () => this.allocate(`parallel-deploy-${crypto.randomUUID()}`)),
  );
  const allocated: AllocatedDevice[] = [];
  const errors: Error[] = [];
  for (const r of settled) {
    if (r.status === 'fulfilled') allocated.push(r.value);
    else errors.push(r.reason instanceof Error ? r.reason : new Error(String(r.reason)));
  }
  if (allocated.length < count) {
    // Release the partials and throw aggregated error — caller surfaces 503.
    for (const d of allocated) await this.release(d.id).catch(() => {});
    throw new AggregateError(errors, `allocateMany: requested ${count}, allocated ${allocated.length}`);
  }
  return allocated;
}
```

import type { PoolModule, AllocatedDevice } from '../../pool/index.js';

export interface ParallelDeployJobInput {
  jobId: string;
  platform: 'android' | 'ios';
  parallelism: number;
  artifactPath: string; // path on disk to .apk or .ipa
  maestroFlow: string;
  envVars?: Record<string, string>;
}

export interface ParallelDeployResult {
  jobId: string;
  deviceCount: number;
  devices: Array<{
    deviceId: string;
    status: 'passed' | 'failed';
    error?: string;
    durationMs?: number;
  }>;
}

export async function runParallelDeploy(
  deps: { pool: PoolModule; installAndLaunch: (device: AllocatedDevice, artifact: string, flow: string, env?: Record<string,string>) => Promise<void>; logger: pino.Logger },
  job: ParallelDeployJobInput,
): Promise<ParallelDeployResult> {
  // Pre-allocate N devices via the new PoolManager.allocateMany. Pool will
  // throw if the partial allocation cannot fill the request (handled at route
  // layer before this is called, but defensive throw on race).
  const devices = await deps.pool.allocateMany(job.platform, job.parallelism);

  // Build artifact already uploaded once via existing multipart upload (job.artifactPath
  // points to it). Skip rebuild — that's the "Build Once" half.

  const perDeviceResults = await Promise.allSettled(
    devices.map(async (d) => {
      const start = Date.now();
      await deps.installAndLaunch(d, job.artifactPath, job.maestroFlow, job.envVars);

  };
}
