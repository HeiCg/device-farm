import { exec } from 'node:child_process';
import { promisify } from 'node:util';
const execAsync = promisify(exec);

export async function extractSymbols(executablePath: string): Promise<string[]> {
  // `nm -gU` shows defined external symbols only (filters SwiftUI internals that are dead-stripped).
  const { stdout } = await execAsync(`nm -gU "${executablePath}"`, { maxBuffer: 64 * 1024 * 1024 });
  return stdout.split('\n').map(l => l.trim().split(/\s+/).at(-1) ?? '').filter(Boolean);
}

export function moduleNameFromSymbol(symbol: string): string | null {
  // Swift mangled symbols start with $s or _$s; module is the first %d-prefixed length-string.
  // Heuristic for filter: if module is one of {SwiftUI, UIKitCore, Foundation, CoreFoundation},
  // exclude from forbidden-symbol rule matches (Pitfall 10).
  const m = symbol.match(/^_?\$s(\d+)([A-Za-z_]+)/);
  if (!m) return null;
  const len = parseInt(m[1], 10);
  return m[2].slice(0, len);
}
```

import rulePack from '../rules/__data__/forbidden-symbols.json' assert { type: 'json' };
import { applyInfoPlistRules } from './rules/ios-info-plist.js';
import { applyPrivacyManifestRules } from './rules/ios-privacy-manifest.js';
import { applyForbiddenSymbolRules } from './rules/ios-forbidden-symbols.js';
import type { ParsedIpa } from './parsers/ipa.js';

export interface RuleFinding {
  rule_id: string;
  severity: 'blocker' | 'warning';
  message: string;
  remediation?: string;
}

export interface PreflightReport {
  status: 'pass' | 'pass_with_warnings' | 'blocked';
  rules_version: string;
  platform: 'ios' | 'android';
  blockers: RuleFinding[];
  warnings: RuleFinding[];
}

export function runRules(bundle: ParsedIpa, platform: 'ios' | 'android'): PreflightReport {
  if (platform !== 'ios') {
    return {
      status: 'pass_with_warnings',
      rules_version: rulePack.rules_updated,
      platform,
      blockers: [],
      warnings: [{ rule_id: 'ANDROID-COMING-V31', severity: 'warning', message: 'Android preflight rules ship in v3.1' }],
    };
  }

  const findings: RuleFinding[] = [
    ...applyInfoPlistRules(bundle, rulePack),
    ...applyPrivacyManifestRules(bundle, rulePack),
    ...applyForbiddenSymbolRules(bundle, rulePack),
  ];

  const blockers = findings.filter(f => f.severity === 'blocker');
  const warnings = findings.filter(f => f.severity === 'warning');
  return {
    status: blockers.length > 0 ? 'blocked' : warnings.length > 0 ? 'pass_with_warnings' : 'pass',
    rules_version: rulePack.rules_updated,
    platform,
    blockers,
    warnings,
  };
}
```

**server/preflight/internal/module.ts:**
```typescript
export interface PreflightModule {
  repo: PreflightRepo;
  events: { completed(runId: string, report: PreflightReport): void };
  scan(input: { filePath: string; filename: string; platform: 'ios' | 'android' }): Promise<{ runId: string; report: PreflightReport }>;
}

export function createPreflightModule(deps: { fastify: FastifyInstance; db: Database; logger: pino.Logger; busFactory: BusFactory }): PreflightModule {
  const repo = createRepo(deps.db);
  const bus = deps.busFactory<PreflightRegistry>();
  const emit = makePreflightEmitters(bus);

  return {
    repo,
    events: { completed: (runId, report) => emit.completed({ preflightRunId: runId, status: report.status, blockerCount: report.blockers.length, warningCount: report.warnings.length }) },
    async scan({ filePath, filename, platform }) {
      const ext = path.extname(filename).toLowerCase();
      const bundle = ext === '.ipa' ? await parseIPA(filePath) : await parseAPK(filePath);
      const report = runRules(bundle, platform);
      const row = await repo.create({
        platform,
        filename,
        fileSizeBytes: (await fs.stat(filePath)).size,
        rulesVersion: report.rules_version,
        status: report.status,
        blockers: report.blockers,
        warnings: report.warnings,
      });
      emit.completed({ preflightRunId: row.id, status: report.status, blockerCount: report.blockers.length, warningCount: report.warnings.length });
      return { runId: row.id, report };
    },
  };
}
```

import { z } from 'zod';

export function registerPreflightRoutes(app: FastifyInstance, deps: { module: PreflightModule }) {
  app.post('/api/preflight', {
    schema: {
      consumes: ['multipart/form-data'],
      response: {
        200: preflightReportSchema,
        413: z.object({ type: z.string(), title: z.string(), status: z.literal(413) }),
        400: z.object({ type: z.string(), title: z.string(), status: z.literal(400) }),
      },
    },
  }, async (req, reply) => {
    const file = await req.file();
    if (!file) { reply.code(400); return { type: 'about:blank#missing-file', title: 'No file uploaded', status: 400 }; }
    const ext = path.extname(file.filename).toLowerCase();
    if (ext !== '.ipa' && ext !== '.apk') { reply.code(400); return { type: 'about:blank#unsupported', title: 'Only .ipa or .apk accepted', status: 400 }; }

    const tmpPath = path.join(os.tmpdir(), `preflight-${crypto.randomUUID()}${ext}`);
    await pipeline(file.file, fs.createWriteStream(tmpPath));
    try {
      const platform = ext === '.ipa' ? 'ios' : 'android';
      const result = await deps.module.scan({ filePath: tmpPath, filename: file.filename, platform });
      return result.report;
    } catch (err) {
      if (err instanceof Error && err.message.includes('Invalid zip')) {
        reply.code(400);
        return { type: 'about:blank#malformed-archive', title: 'Could not parse archive', status: 400 };
      }
      throw err;
    } finally {
      await fs.promises.rm(tmpPath, { force: true });
    }
  });

  app.get('/api/preflight/:id', async (req, reply) => {
    const row = await deps.module.repo.getById(req.params.id);
    if (!row) { reply.code(404); return { type: 'about:blank#not-found', title: 'Preflight run not found', status: 404 }; }
    return row;
  });
}
