```typescript
export interface CommenterDeps {
  getOctokitForInstallation: (installationId: number) => Promise<Octokit>;
  lookupCommentId: (input: { installationId: number; prNumber: number }) => Promise<{ commentId: number } | null>;
  saveCommentId: (input: { installationId: number; prNumber: number; commentId: number }) => Promise<void>;
  baseUrl: string;
  logger: pino.Logger;
}

export function createGithubCommenter(deps: CommenterDeps) {
  return {
    async upsert(input: {
      installationId: number;
      owner: string;
      repo: string;
      prNumber: number;
      status: 'running' | 'passed' | 'failed' | 'cancelled';
      jobId: string;
      deviceLabel: string;
      duration?: string;
      screenshotsSignedUrls: string[];
    }) {
      const octokit = await deps.getOctokitForInstallation(input.installationId);
      const body = renderComment({
        status: input.status,
        jobId: input.jobId,
        deviceLabel: input.deviceLabel,
        duration: input.duration,
        screenshotsSignedUrls: input.screenshotsSignedUrls,
        dashboardUrl: `${deps.baseUrl}/jobs/${input.jobId}`,
      });
      const existing = await deps.lookupCommentId({ installationId: input.installationId, prNumber: input.prNumber });
      if (existing) {
        await octokit.request('PATCH /repos/{owner}/{repo}/issues/comments/{comment_id}', {
          owner: input.owner, repo: input.repo, comment_id: existing.commentId, body,
        });
      } else {
        const res = await octokit.request('POST /repos/{owner}/{repo}/issues/{issue_number}/comments', {
          owner: input.owner, repo: input.repo, issue_number: input.prNumber, body,
        });
        await deps.saveCommentId({ installationId: input.installationId, prNumber: input.prNumber, commentId: res.data.id });
      }
    },
  };
}
```

```typescript
export interface WebhookOutcome {
  kind: 'triggered' | 'ignored' | 'deferred';
  reason?: string;
  runId?: string;
}

export function createGithubWebhookHandler(deps: {
  triggerRun: (input: {...}) => Promise<{runId: string}>;
  logger: pino.Logger;
}) {
  return {
    async handle(event: string, body: unknown): Promise<WebhookOutcome> {
      if (event !== 'pull_request') return { kind: 'ignored', reason: 'event-not-handled' };
      const parsed = pullRequestEventSchema.safeParse(body);
      if (!parsed.success) return { kind: 'ignored', reason: 'parse-failed' };
      const evt = parsed.data;
      if (!['opened', 'synchronize', 'reopened'].includes(evt.action)) {
        return { kind: 'ignored', reason: `action-${evt.action}-not-handled` };
      }
      // delegate to triggerRun
      const r = await deps.triggerRun({
        installationId: evt.installation.id,
        owner: evt.repository.owner.login,
        repo: evt.repository.name,
        prNumber: evt.pull_request.number,
        prUrl: evt.pull_request.html_url,
        headSha: evt.pull_request.head.sha,
        headRef: evt.pull_request.head.ref,
      });
      return { kind: 'triggered', runId: r.runId };
    },
  };
}
```

```typescript
export interface GithubModule {
  webhookSecret: string;
  handler: ReturnType<typeof createGithubWebhookHandler>;
  commenter: ReturnType<typeof createGithubCommenter>;
  app: App;
}

export async function createGithubModule(deps: { fastify: FastifyInstance; logger: pino.Logger }): Promise<GithubModule | null> {
  const cfg = deps.fastify.config.github;
  if (!cfg) { deps.logger.info('github config absent — integration disabled'); return null; }

  const app = createGithubApp({ appId: cfg.app_id, privateKey: cfg.private_key, webhookSecret: cfg.webhook_secret });

  const baseUrl = `http://${deps.fastify.config.server.host}:${deps.fastify.config.server.port}`;

  const commenter = createGithubCommenter({
    getOctokitForInstallation: (installationId) => getInstallationOctokit(app, installationId),
    lookupCommentId: async ({ installationId, prNumber }) => {
      const [row] = await deps.fastify.db.select({ id: schema.pipelineRuns.githubPrCommentId })
        .from(schema.pipelineRuns)
        .where(and(
          eq(schema.pipelineRuns.githubInstallationId, installationId),
          eq(schema.pipelineRuns.githubPrId, String(prNumber)),
          isNotNull(schema.pipelineRuns.githubPrCommentId),
        ))
        .orderBy(desc(schema.pipelineRuns.startedAt))
        .limit(1);
      return row?.id ? { commentId: Number(row.id) } : null;
    },
    saveCommentId: async ({ installationId, prNumber, commentId }) => {
      await deps.fastify.db.update(schema.pipelineRuns)
        .set({ githubPrCommentId: String(commentId) })
        .where(and(
          eq(schema.pipelineRuns.githubInstallationId, installationId),
          eq(schema.pipelineRuns.githubPrId, String(prNumber)),
          isNull(schema.pipelineRuns.githubPrCommentId),
        ));
    },
    baseUrl,
    logger: deps.logger,
  });

  const prRunService = createGithubPrRunService({ fastify: deps.fastify, integrations: cfg.pr_integrations, logger: deps.logger });
  const handler = createGithubWebhookHandler({ triggerRun: (req) => prRunService.triggerRun(req), logger: deps.logger });

  return { webhookSecret: cfg.webhook_secret, handler, commenter, app };
}
```

```typescript
import { verify } from '@octokit/webhooks-methods';

export function registerGithubRoutes(app: FastifyInstance, deps: { webhookSecret: string; handler: { handle(event: string, body: unknown): Promise<WebhookOutcome> }; logger: pino.Logger }) {
  // CRITICAL: Pitfall 1 — Fastify default JSON parser destroys raw bytes.
  // Capture buffer pre-parse so HMAC signature matches GitHub's signed payload.
  app.addContentTypeParser('application/json', { parseAs: 'buffer' }, (req, body, done) => {
    try {
      const parsed = JSON.parse(body.toString('utf8'));
      done(null, { __raw: body, __parsed: parsed });
    } catch (err) {
      done(err as Error, undefined);
    }
  });

  app.post('/webhooks/github', async (req, reply) => {
    const sig = req.headers['x-hub-signature-256'] as string | undefined;
    const event = req.headers['x-github-event'] as string | undefined;
    const delivery = req.headers['x-github-delivery'] as string | undefined;

    if (!sig || !event || !delivery) {
      reply.code(400);
      return { error: 'missing required headers' };
    }

    const body = req.body as { __raw: Buffer; __parsed: unknown };
    const ok = await verify(deps.webhookSecret, body.__raw.toString('utf8'), sig);
    if (!ok) {
      deps.logger.warn({ event, delivery }, 'github webhook: invalid signature');
      reply.code(401);
      return { error: 'invalid signature' };
    }

    const outcome = await deps.handler.handle(event, body.__parsed);
    reply.code(200);
    return outcome;
  });
}
```

```typescript
async function githubPlugin(fastify: FastifyInstance): Promise<void> {
  const mod = await createGithubModule({ fastify, logger: fastify.log });
  fastify.decorate('githubModule', mod);

  if (mod) {
    registerGithubRoutes(fastify, { webhookSecret: mod.webhookSecret, handler: mod.handler, logger: fastify.log });
    fastify.log.info('GitHub PR-bot plugin registered');

    // onReady-deferred bus subscription, mirroring server/azure/plugin.ts:39-141.
    fastify.addHook('onReady', async () => {
      const bus = fastify.pipelinesModule.bus;
      const commenter = mod.commenter;
      const log = fastify.log;

      async function loadGithubRunContext(runId: string) {
        const [row] = await fastify.db.select({
          githubInstallationId: schema.pipelineRuns.githubInstallationId,
          githubPrId: schema.pipelineRuns.githubPrId,
          githubRepoOwner: schema.pipelineRuns.githubRepoOwner,
          githubRepoName: schema.pipelineRuns.githubRepoName,
          startedAt: schema.pipelineRuns.startedAt,
        }).from(schema.pipelineRuns).where(eq(schema.pipelineRuns.id, runId)).limit(1);

        if (!row || !row.githubInstallationId || !row.githubPrId) return null;
        return {
          installationId: row.githubInstallationId,
          prNumber: Number(row.githubPrId),
          owner: row.githubRepoOwner ?? '',
          repo: row.githubRepoName ?? '',
          startedAt: row.startedAt ?? new Date(),
        };
      }

      async function upsertComment(runId: string, status: 'running' | 'passed' | 'failed' | 'cancelled') {
        try {
          const ctx = await loadGithubRunContext(runId);
          if (!ctx) return;
          // Look up artifact URLs for this run via fastify.artifactsModule (best-effort)
          const screenshots = await fastify.artifactsModule?.listSignedUrlsForRun?.(runId, 'screenshot') ?? [];
          await commenter.upsert({
            installationId: ctx.installationId,
            owner: ctx.owner,
            repo: ctx.repo,
            prNumber: ctx.prNumber,
            status,
            jobId: runId,
            deviceLabel: 'Device Farm',
            screenshotsSignedUrls: screenshots,
          });
        } catch (err) {
          log.error({ err, runId, status }, 'github commenter: upsert failed');
        }
      }

      bus.on(PIPELINE_EVENT_NAMES.RUN_STARTED, (payload) => void upsertComment(payload.runId, 'running'));
      bus.on(PIPELINE_EVENT_NAMES.RUN_COMPLETED, (payload) => {
        const status = payload.status === 'passed' ? 'passed' : payload.status === 'cancelled' ? 'cancelled' : 'failed';
        void upsertComment(payload.runId, status);
      });
      bus.on(PIPELINE_EVENT_NAMES.RUN_FAILED, (payload) => {
        const status = payload.reason === 'cancelled' ? 'cancelled' : 'failed';
        void upsertComment(payload.runId, status);
      });

      log.info('GitHub PR-bot commenter wired to pipeline event bus');
    });
  }
}

export default fp(githubPlugin, { name: 'github-plugin', dependencies: ['config', 'db', 'pipelines-plugin'] });
```
