    ```typescript
     * Pipelines scheduler — boss.schedule-backed (Phase 25 Plan 25-02).
     *
     * REPLACES the node-cron implementation. ZERO node-cron imports remain.
     * Per-pipelineSchedules row maps to ONE pg-boss schedule, disambiguated by
     * the row's UUID via the `key` parameter (NOT singletonKey, NOT pipelineId
     * — see RESEARCH §Pattern 1 Schedule Identity Decision).
     *
     * Lifecycle:
     *   - reconcileSchedules() runs at boot via Plan 25-03 createPipelinesModule.
     *     Mirror of pgboss.schedule from pipelineSchedules; orphan cleanup.
     *   - addSchedule(scheduleId) called from routes on schedule create/update.
     *   - removeSchedule(scheduleId) called from routes on schedule delete (and
     *     from service.deletePipeline cascade — see Pitfall 3).
     *
     * Schedule fires invoke the queue worker (registered in Plan 25-03 factory),
     * NOT a callback held in this class. The schedule-fire → run-start chain
     * lives in queue.ts (worker) + service.ts (run starter).
     */

    import { eq } from 'drizzle-orm';
    import type pino from 'pino';
    import type PgBoss from 'pg-boss';
    import type { Database } from '../db/index.js';
    import * as schema from '../db/schema.js';
    import {
      PIPELINE_SCHEDULED_EXECUTE_QUEUE_NAME,
      upsertPipelineSchedule,
      removePipelineSchedule,
    } from './queue.js';

    export interface PipelineSchedulerDeps {
      db: Database;
      logger: pino.Logger;
      boss: PgBoss;
      /** Retained for plugin.ts back-compat until Plan 25-03 deletes plugin.ts; unused here. */
      pipelineService?: unknown;
    }

    export class PipelineScheduler {
      private readonly db: Database;
      private readonly logger: pino.Logger;
      private readonly boss: PgBoss;

      constructor(deps: PipelineSchedulerDeps) {
        this.db = deps.db;
        this.logger = deps.logger.child({ component: 'pipeline-scheduler' });
        this.boss = deps.boss;
      }

      /**
       * Reconcile pgboss.schedule with pipelineSchedules. Idempotent — safe on every boot.
       * (a) For every enabled pipelineSchedules row, call upsertPipelineSchedule.
       * (b) For every pgboss schedule whose key is NOT in the app-side row set, unschedule (orphan cleanup).
       *
       * Replaces the legacy `start()` method. Synchronous wrt onReady (Open Question 2:
       * with < 100 pipelines, sub-second).
       */
      async reconcileSchedules(): Promise<void> {
        const rows = await this.db
          .select()
          .from(schema.pipelineSchedules)
          .where(eq(schema.pipelineSchedules.enabled, true));

        for (const row of rows) {
          await upsertPipelineSchedule({
            boss: this.boss,
            scheduleId: row.id,
            pipelineId: row.pipelineId,
            cronExpression: row.cronExpression,
            variables: (row.variables as Record<string, string>) ?? {},
          });
        }

        // Orphan cleanup: any pgboss.schedule key not in the app-side enabled set
        const bossSchedules = await this.boss.getSchedules();
        const appKeys = new Set(rows.map((r) => r.id));
        for (const s of bossSchedules) {
          // pg-boss Schedule object: { name, cron, key, tz, data }
          if (s.name === PIPELINE_SCHEDULED_EXECUTE_QUEUE_NAME && s.key && !appKeys.has(s.key)) {
            this.logger.warn({ key: s.key }, 'Orphan pgboss schedule — unscheduling');
            await this.boss.unschedule(PIPELINE_SCHEDULED_EXECUTE_QUEUE_NAME, s.key);
          }
        }

        this.logger.info({ count: rows.length }, 'Pipeline scheduler reconciled');
      }

      /**
       * Add or update a schedule (called from routes.ts on POST/PUT). Idempotent.
       * Throws (via boss.schedule → CronExpressionParser) if cronExpression is invalid —
       * BEHAVIOUR CHANGE from legacy node-cron's silent-log-and-skip (lines 85-88).
       */
      async addSchedule(scheduleId: string): Promise<void> {
        const [row] = await this.db
          .select()
          .from(schema.pipelineSchedules)
          .where(eq(schema.pipelineSchedules.id, scheduleId));

        if (!row || !row.enabled) {
          // Disabled or missing — ensure no pg-boss schedule exists
          await removePipelineSchedule({ boss: this.boss, scheduleId });
          return;
        }

        await upsertPipelineSchedule({
          boss: this.boss,
          scheduleId: row.id,
          pipelineId: row.pipelineId,
          cronExpression: row.cronExpression,
          variables: (row.variables as Record<string, string>) ?? {},
        });
      }

      /** Remove a schedule (called from routes.ts on DELETE + service.deletePipeline cascade). */
      async removeSchedule(scheduleId: string): Promise<void> {
        await removePipelineSchedule({ boss: this.boss, scheduleId });
      }

      /**
       * Inspection helper retained for back-compat / debug routes. Reads pg-boss
       * source-of-truth (NOT an in-memory Map — that's deleted).
       */
      async getActiveSchedules(): Promise<Array<{ id: string; pipelineId: string; cron: string }>> {
        const bossSchedules = await this.boss.getSchedules();
        return bossSchedules
          .filter((s) => s.name === PIPELINE_SCHEDULED_EXECUTE_QUEUE_NAME && s.key)
          .map((s) => ({
            id: s.key as string,
            pipelineId: ((s.data as { pipelineId?: string } | undefined)?.pipelineId) ?? 'unknown',
            cron: s.cron,
          }));
      }

      /**
       * No-op. pg-boss schedules persist across restarts; the worker is offWork'd
       * by createPipelinesModule.shutdown() in Plan 25-03.
       */
      stop(): void {
        this.logger.debug('PipelineScheduler.stop — no-op (pg-boss schedules persist; worker offWork'd by module shutdown)');
      }

      /**
       * Back-compat alias for plugin.ts pre-25-03; calls reconcileSchedules.
       * Removed when plugin.ts is rewritten in Plan 25-03.
       */
      async start(): Promise<void> {
        await this.reconcileSchedules();
      }
    }
    ```
