    ```typescript
    import fp from 'fastify-plugin';
    import { PgBoss } from 'pg-boss';
    import { randomUUID } from 'node:crypto';
    import { requestContext, asyncLocalStorage } from '@fastify/request-context';
    import { QUEUE_NAMES, isValidQueueName } from './names.js';

    declare module 'fastify' {
      interface FastifyInstance {
        boss: PgBoss;
        queue: {
          send: (name: string, data: unknown, opts?: Record<string, unknown>) => Promise<string | null>;
          work: <T = unknown>(
            name: string,
            handler: (data: T, jobId: string) => Promise<void> | void,
          ) => Promise<string>;
          schedule: (name: string, cron: string, data?: unknown, opts?: Record<string, unknown>) => Promise<void>;
        };
      }
    }

    export default fp(async function queuePlugin(fastify) {
      const connectionString = fastify.config.database_url;
      const boss = new PgBoss({
        connectionString,
        schema: 'pgboss',
        application_name: 'device-farm-server',
      });

      boss.on('error', (err) => fastify.log.error({ err }, 'pg-boss internal error'));

      await boss.start();
      fastify.log.info('pg-boss started (schema=pgboss)');

      // Phase 15: register the demo queue used by spikes + integration tests.
      // Future phases register their own queues in their modules' queue.ts.
      await boss.createQueue(QUEUE_NAMES.DEMO, { retryLimit: 1, retryBackoff: true, retryDelay: 30 });

      fastify.decorate('boss', boss);

      fastify.decorate('queue', {
        async send(name, data, opts = {}) {
          if (!isValidQueueName(name)) throw new Error(`Invalid queue name: ${name}`);
          const correlationId = (requestContext.get('correlationId') as string | null) ?? randomUUID();
          const causationId   = (requestContext.get('currentEventId') as string | null) ?? null;
          const actor         = (requestContext.get('actor') as string | null) ?? 'system';
          return boss.send(name, { correlationId, causationId, actor, payload: data }, opts);
        },
        async work<T = unknown>(name: string, handler: (data: T, jobId: string) => Promise<void> | void) {
          if (!isValidQueueName(name)) throw new Error(`Invalid queue name: ${name}`);
          return boss.work(name, async (jobs: any) => {
            const job = Array.isArray(jobs) ? jobs[0] : jobs;
            const store = new Map<string, unknown>([
              ['correlationId', job.data?.correlationId ?? randomUUID()],
              ['currentEventId', job.data?.causationId ?? null],
              ['actor', job.data?.actor ?? 'cron'],
            ]);
            await asyncLocalStorage.run(store, async () => {
              await handler(job.data?.payload as T, job.id);
            });
          });
        },
        async schedule(name, cron, data = {}, opts = {}) {
          if (!isValidQueueName(name)) throw new Error(`Invalid queue name: ${name}`);
          return boss.schedule(name, cron, {
            correlationId: randomUUID(),
            causationId: null,
            actor: 'cron',
            payload: data,
          }, opts);
        },
      });

      fastify.addHook('onClose', async () => {
        fastify.log.info('pg-boss: stopping (graceful, timeout=30s)...');
        await boss.stop({ graceful: true, timeout: 30_000, destroy: false });
        fastify.log.info('pg-boss: stopped');
      });
    }, { name: 'queue', dependencies: ['db', 'correlation'] });
    ```

    ```typescript
    import { buildApp } from '../../server/index.js';
    import { QUEUE_NAMES } from '../../server/queue/names.js';

    async function main() {
      const app = await buildApp();
      await app.listen({ port: 0, host: '127.0.0.1' });

      // Let workers pick up a few before we close

      const start = Date.now();
      await app.close();
      const elapsed = Date.now() - start;
      console.log(`close took ${elapsed}ms`);
    }

    main().catch((err) => {
      console.error(err);
      process.exit(1);
    });
    ```
