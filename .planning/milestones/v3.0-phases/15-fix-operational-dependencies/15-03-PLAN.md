    ```typescript
    import fp from 'fastify-plugin';
    import fastifyRequestContext from '@fastify/request-context';
    import { randomUUID } from 'node:crypto';

    export default fp(async function correlationPlugin(fastify) {
      await fastify.register(fastifyRequestContext, {
        hook: 'onRequest',
        defaultStoreValues: {
          correlationId: null,
          currentEventId: null,
          actor: 'anonymous',
        },
      });

      fastify.addHook('onRequest', async (req, reply) => {
        const raw = req.headers['x-correlation-id'];
        const ok = typeof raw === 'string'
          && raw.length > 0 && raw.length < 128
          && /^[\x20-\x7E]+$/.test(raw);
        const correlationId = ok ? raw : randomUUID();
        req.requestContext.set('correlationId', correlationId);
        reply.header('x-correlation-id', correlationId);
      });
    }, { name: 'correlation', dependencies: [] });
    ```

    ```typescript
    import fp from 'fastify-plugin';
    import { asyncLocalStorage } from '@fastify/request-context';

    export const alsMixin = () => {
      const store = asyncLocalStorage.getStore();
      if (!store) return {};
      const out: Record<string, unknown> = {};
      // request-context stores as Map<string, unknown>
      const cid   = (store as Map<string, unknown>).get?.('correlationId');
      const cause = (store as Map<string, unknown>).get?.('currentEventId');
      const actor = (store as Map<string, unknown>).get?.('actor');
      if (cid)   out.correlationId = cid;
      if (cause) out.causationId   = cause;
      if (actor) out.actor         = actor;
      return out;
    };

    export default fp(async function telemetryPlugin(fastify) {
      // The Pino mixin itself is wired at Fastify() construction time (server/index.ts).
      // This plugin is a seam for future metrics decorations and a single import for `alsMixin`.

      });
    }, { name: 'telemetry', dependencies: ['correlation'] });
    ```
