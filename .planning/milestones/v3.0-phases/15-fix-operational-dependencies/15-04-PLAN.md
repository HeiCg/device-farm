    ```typescript
    import { EventEmitter } from 'node:events';
    import type { EventRegistry, PayloadOf } from './types.js';

    export class TypedBus<R extends EventRegistry> {
      private readonly ee = new EventEmitter();

      constructor(public readonly registry: R) {
        this.ee.setMaxListeners(50);
      }

      on<T extends keyof R & string>(
        type: T,
        handler: (payload: PayloadOf<R, T>) => void | Promise<void>,
      ): () => void {
        const wrapped = (payload: unknown) => handler(payload as PayloadOf<R, T>);
        this.ee.on(type, wrapped);
        return () => this.ee.off(type, wrapped);
      }

      emit<T extends keyof R & string>(type: T, payload: PayloadOf<R, T>): void {
        const entry = this.registry[type];
        if (!entry) throw new Error(`Unknown event type: ${String(type)}`);
        const parsed = entry.schema.parse(payload);
        this.ee.emit(type, parsed);
      }
    }
    ```

    import { randomUUID } from 'node:crypto';
    import { requestContext } from '@fastify/request-context';
    import { envelopeSchema, type Envelope } from '../events/envelope.js';
    import type { EventRegistry, PayloadOf } from './types.js';
    import type { TypedBus } from './bus.js';

    type HelperOpts = { actor?: string };

    export function createEventHelpers<R extends EventRegistry>(
      bus: TypedBus<R>,
      onEmit?: (envelope: Envelope) => void,  // hook for persistence middleware
    ) {
      return <T extends keyof R & string>(type: T) =>
        (aggregateId: string, payload: PayloadOf<R, T>, opts: HelperOpts = {}): Envelope => {
          const entry = bus.registry[type];
          if (!entry) throw new Error(`Unknown event type: ${String(type)}`);

          const envelope = envelopeSchema.parse({
            id: randomUUID(),
            type,
            v: 1,
            correlationId: (requestContext.get('correlationId') as string | null) ?? randomUUID(),
            causationId: (requestContext.get('currentEventId') as string | null) ?? null,
            occurredAt: new Date().toISOString(),
            aggregateType: entry.aggregateType ?? type.split('.')[0],
            aggregateId,
            actor: opts.actor ?? (requestContext.get('actor') as string | null) ?? 'anonymous',
            payload,
          });

          bus.emit(type, payload);
          onEmit?.(envelope);
          return envelope;
        };
    }

    **Envelope-forwarding approach (PINNED — side-channel `<type>.envelope` emit):**

    `bus.on` only delivers the payload (per task 4.1), but `onPersisted` subscribers need access to the envelope (for `causationId` propagation + access to headers). We use a **side-channel emit** on a parallel event name `<type>.envelope` to forward the envelope to subscribers registered via `onPersisted`.

    Rationale (do not deviate):
    - The WeakMap<payload, envelope> alternative ties correctness to JS reference equality, which is fragile across event-loop boundaries (promise ticks, microtasks, handler copies) and invisible when debugging.
    - Side-channel emits are explicit and greppable in subscriber code (`grep '\\.envelope'` shows every consumer).
    - Avoids modifying the public `TypedBus.emit` signature, keeping the API surface small.

    Inside `createEventHelpers.onEmit` the persistence middleware ALREADY receives the envelope as an argument, so DB persistence does not need the side channel. The side channel exists solely for `onPersisted` subscribers to receive the envelope for causation/header propagation.

    Create `server/bus/plugin.ts`:

    ```typescript
    import fp from 'fastify-plugin';
    import { TypedBus } from './bus.js';
    import { createEventHelpers } from './helpers.js';
    import { demoRegistry } from '../events/registry.js';
    import { events as eventsTable } from '../db/schema.js';
    import type { Envelope } from '../events/envelope.js';
    import { asyncLocalStorage } from '@fastify/request-context';

    declare module 'fastify' {
      interface FastifyInstance {
        bus: TypedBus<typeof demoRegistry>;
        emit: ReturnType<typeof createEventHelpers<typeof demoRegistry>>;
        onPersisted: <T extends keyof typeof demoRegistry & string>(
          type: T,
          handler: (envelope: Envelope) => void | Promise<void>,
        ) => () => void;
      }
    }

    export default fp(async function busPlugin(fastify) {
      const bus = new TypedBus(demoRegistry);

      // Envelope-forwarding strategy: side-channel emit on `<type>.envelope`.
      // When an emit helper stamps an envelope, it also fires a second event
      // `<type>.envelope` carrying the envelope. onPersisted subscribers listen
      // on this side channel so they receive the envelope (not just the payload),
      // which lets them propagate causationId into ALS for nested emits.
      //
      // Rationale: explicit + greppable. WeakMap<payload,envelope> alternative
      // was rejected for ref-equality fragility across event-loop boundaries.

      // persistence middleware — only for `persisted: true` events
      const persistEnvelope = async (envelope: Envelope) => {
        // Fire the side-channel envelope event FIRST so onPersisted subscribers
        // observe the envelope synchronously on the same emit call.
        (bus as any).ee.emit(`${envelope.type}.envelope`, envelope);

        const entry = demoRegistry[envelope.type as keyof typeof demoRegistry];
        if (!entry || !entry.persisted) return;
        try {
          await fastify.db.insert(eventsTable).values({
            id: envelope.id,
            eventType: envelope.type,
            eventVersion: envelope.v,
            correlationId: envelope.correlationId,
            causationId: envelope.causationId ?? undefined,
            aggregateType: envelope.aggregateType,
            aggregateId: envelope.aggregateId,
            payload: envelope.payload as any,
            occurredAt: new Date(envelope.occurredAt),
            actor: envelope.actor,
          });
        } catch (err) {
          fastify.log.error({ err, envelope }, 'Failed to persist event');
          // do NOT rethrow — persistence is best-effort at the bus layer
        }
      };

      const emit = createEventHelpers(bus, persistEnvelope);

      // subscriber wrapper: subscribes on the `<type>.envelope` side channel,
      // sets currentEventId in ALS before handler runs so nested emits auto-
      // populate causationId.
      const onPersisted: FastifyInstance['onPersisted'] = (type, handler) => {
        const sideChannel = `${type}.envelope`;
        const listener = async (envelope: Envelope) => {
          const store = asyncLocalStorage.getStore() as Map<string, unknown> | undefined;
          const nextStore = new Map(store ?? []);
          nextStore.set('currentEventId', envelope.id);
          return asyncLocalStorage.run(nextStore, () => handler(envelope));
        };
        (bus as any).ee.on(sideChannel, listener);
        return () => (bus as any).ee.off(sideChannel, listener);
      };

      fastify.decorate('bus', bus);
      fastify.decorate('emit', emit);
      fastify.decorate('onPersisted', onPersisted);

      fastify.addHook('onClose', async () => {
        (bus as any).ee?.removeAllListeners?.();
      });
    }, { name: 'event-bus', dependencies: ['db', 'correlation'] });
    ```
