    import { asyncLocalStorage } from '@fastify/request-context';
    import { buildApp } from '<path/to/test-helper>';
    import type { Envelope } from '../../events/envelope.js';

    const SKIP_DB = !process.env.DEVICE_FARM_TEST_DB;

    describe.skipIf(SKIP_DB)('maestro correlation.spec — ALS correlationId thread', () => {
      let app: Awaited<ReturnType<typeof buildApp>>;

        app = await buildApp();

        const envelopes: Envelope[] = [];
        (app.maestroModule.bus as any).ee.on(
          'maestro.device-info.collected.envelope',

     * Phase 24 / Plan 24-04 — Lifecycle ownership grep-guards.
     *
     * Pure readFileSync structural assertions proving SC1 architectural invariants:
     *
     *   (a) Old import path 'pool/device-info-collector' is dead after git mv
     *   (b) Imperative metadata-collection loop in plugin.ts is GONE
     *   (c) onReady deferral is present (1 hit; the registerSubscribers call)

     *   (e) DeviceInfoCollector.collect calls relocated to subscribers.ts (not deleted)
     *
     * No DB. No fastify. Pure readFileSync + regex.
     */
    import { readFileSync, readdirSync, statSync } from 'node:fs';
    import { join } from 'node:path';
    import { describe, it, expect } from 'vitest';

    function readSourceFile(p: string): string {
      return readFileSync(p, 'utf8');
    }

    function* walkServerSrc(dir: string): Generator<string> {
      for (const entry of readdirSync(dir)) {
        const full = join(dir, entry);
        const st = statSync(full);
        if (st.isDirectory()) {
          if (entry === '__tests__' || entry === 'node_modules') continue;
          yield* walkServerSrc(full);
        } else if (entry.endsWith('.ts') && !entry.endsWith('.d.ts')) {
          yield full;
        }
      }
    }

    describe('Phase 24 lifecycle-ownership grep-guards', () => {
      it('(a) No source file imports from pool/device-info-collector (old path dead)', () => {
        const offenders: string[] = [];
        for (const file of walkServerSrc('server')) {
          const src = readSourceFile(file);
          if (/from\s+['"][^'"]*pool\/device-info-collector/.test(src)) {
            offenders.push(file);
          }
        }
        expect(offenders).toEqual([]);
      });

      it('(b) server/maestro/plugin.ts has zero deviceInfoCollector.collect() calls (loop deleted)', () => {
        const src = readSourceFile('server/maestro/plugin.ts');

        const matches = src.match(/fastify\.addHook\(\s*['"]onReady['"]/g) ?? [];
        expect(matches).toHaveLength(1);
      });

      it('(e) server/index.ts STILL HAS hookExecutor.execute("device.booted") loop (different surface preserved)', () => {
        const src = readSourceFile('server/index.ts');
