<!-- Wire format — BRIEF locks this -->
Server → client (write):
  0x01: avcC SPS+PPS parameter set (one-shot on first IDR)
  0x02: H.264 access unit (length-prefixed NAL units inside, AVCC format)
  0x10: ack (e.g., for control acknowledgements)
  0xFF: error (UTF-8 message payload)

Client → server (read):
  0xC1: touch — payload [f64 BE x_ratio][f64 BE y_ratio][u8 phase][f64 BE pressure][u32 BE touchId] = 29 bytes
  0xC9: quit  — payload empty (length=1, just the kind byte)

Frame format on the wire:
  [u32 BE length][u8 kind][payload bytes]
  where length covers (1 byte kind + payload bytes), matching sim-capture-avcc/main.swift StdoutWriter.

```objc
#import "H264Encoder.h"
#import <VideoToolbox/VideoToolbox.h>
#import <CoreMedia/CoreMedia.h>

@interface DSH264Encoder ()
@property (nonatomic, assign) VTCompressionSessionRef session;
@property (nonatomic, assign) DSEncodedCallback cb;
@property (nonatomic, assign) void *userData;
@property (nonatomic, assign) BOOL paramSetsSent;
@property (nonatomic, assign) int width;
@property (nonatomic, assign) int height;
@end

static void DSEncoderOutputCallback(void *outputCallbackRefCon, void *sourceFrameRefCon,
                                     OSStatus status, VTEncodeInfoFlags infoFlags,
                                     CMSampleBufferRef sampleBuffer) {
  DSH264Encoder *self = (__bridge DSH264Encoder *)outputCallbackRefCon;
  if (status != noErr || !sampleBuffer) return;

  // 1) Extract avcC params on first sample (or after IDR)
  if (!self.paramSetsSent) {
    CMFormatDescriptionRef fmt = CMSampleBufferGetFormatDescription(sampleBuffer);
    size_t spsSize=0, spsCount=0; const uint8_t *spsPtr=NULL;
    size_t ppsSize=0, ppsCount=0; const uint8_t *ppsPtr=NULL;
    CMVideoFormatDescriptionGetH264ParameterSetAtIndex(fmt, 0, &spsPtr, &spsSize, &spsCount, NULL);
    CMVideoFormatDescriptionGetH264ParameterSetAtIndex(fmt, 1, &ppsPtr, &ppsSize, &ppsCount, NULL);

    // Build avcC blob: [u8 0x01][u8 profile][u8 compat][u8 level][u8 0xFC|3=NALU-length-3]
    //                 [u8 0xE0|1=numSPS][u16 BE spsLen][SPS bytes][u8 1=numPPS][u16 BE ppsLen][PPS bytes]
    NSMutableData *avcc = [NSMutableData data];
    uint8_t header[5] = { 0x01, spsPtr[1], spsPtr[2], spsPtr[3], 0xFC | 3 };
    [avcc appendBytes:header length:sizeof(header)];
    uint8_t numSps = 0xE0 | 1;
    [avcc appendBytes:&numSps length:1];
    uint16_t spsLen = htons((uint16_t)spsSize);
    [avcc appendBytes:&spsLen length:2];
    [avcc appendBytes:spsPtr length:spsSize];
    uint8_t numPps = 1;
    [avcc appendBytes:&numPps length:1];
    uint16_t ppsLen = htons((uint16_t)ppsSize);
    [avcc appendBytes:&ppsLen length:2];
    [avcc appendBytes:ppsPtr length:ppsSize];

    self.cb(DSEncodedKindParamSets, avcc.bytes, avcc.length, self.userData);
    self.paramSetsSent = YES;
  }

  // 2) Emit AVCC NALs (length-prefixed). CMBlockBuffer is already AVCC by VT default.
  CMBlockBufferRef bb = CMSampleBufferGetDataBuffer(sampleBuffer);
  size_t totalLen = 0;
  char *dataPtr = NULL;
  CMBlockBufferGetDataPointer(bb, 0, NULL, &totalLen, &dataPtr);
  self.cb(DSEncodedKindAccessUnit, (const uint8_t *)dataPtr, totalLen, self.userData);
}

@implementation DSH264Encoder

- (instancetype)initWithWidth:(int)width height:(int)height fps:(int)fps bitrate:(int)bitrate
              outputCallback:(DSEncodedCallback)cb userData:(void *)userData {
  if ((self = [super init])) {
    _cb = cb;
    _userData = userData;
    _width = width;
    _height = height;
    _paramSetsSent = NO;

    OSStatus s = VTCompressionSessionCreate(kCFAllocatorDefault, width, height,
                                             kCMVideoCodecType_H264, NULL, NULL, NULL,
                                             DSEncoderOutputCallback, (__bridge void *)self,
                                             &_session);
    if (s != noErr) { fprintf(stderr, "VTCompressionSessionCreate failed: %d\n", (int)s); return nil; }
    VTSessionSetProperty(_session, kVTCompressionPropertyKey_RealTime, kCFBooleanTrue);
    VTSessionSetProperty(_session, kVTCompressionPropertyKey_AllowFrameReordering, kCFBooleanFalse);
    VTSessionSetProperty(_session, kVTCompressionPropertyKey_ProfileLevel, kVTProfileLevel_H264_Baseline_AutoLevel);
    VTSessionSetProperty(_session, kVTCompressionPropertyKey_MaxKeyFrameInterval, (__bridge CFTypeRef)@30);
    VTSessionSetProperty(_session, kVTCompressionPropertyKey_AverageBitRate, (__bridge CFTypeRef)@(bitrate));
    VTSessionSetProperty(_session, kVTCompressionPropertyKey_ExpectedFrameRate, (__bridge CFTypeRef)@(fps));
    VTCompressionSessionPrepareToEncodeFrames(_session);
  }
  return self;
}

- (void)encodePixelBuffer:(CVPixelBufferRef)pb forceKeyframe:(BOOL)force {
  NSDictionary *frameProps = force ? @{(NSString *)kVTEncodeFrameOptionKey_ForceKeyFrame: @YES} : nil;
  CMTime pts = CMTimeMakeWithSeconds(CACurrentMediaTime(), 1000000);
  VTCompressionSessionEncodeFrame(_session, pb, pts, kCMTimeInvalid,
                                   (__bridge CFDictionaryRef)frameProps, NULL, NULL);
}

- (void)forceIDR { /* set encode-frame flag on next call */ }
- (void)teardown {
  if (_session) { VTCompressionSessionCompleteFrames(_session, kCMTimeInvalid); VTCompressionSessionInvalidate(_session); CFRelease(_session); _session = NULL; }
}
@end
```

2. `Sources/IpcServer.mm` — implement:
- `DSFramerDecodeOne` — pure function, no I/O. Reads 4-byte BE length, then kind, then payload. Returns NO if `available < 4 + length`. Sets `*consumed = 4 + length` on success.
- `DSIpcServer` — `AF_UNIX` `SOCK_STREAM` socket. `unlink(path)` → `bind` → `listen(fd, 1)` → `accept`. On accepted fd: set `SO_NOSIGPIPE` (setsockopt level=SOL_SOCKET optname=SO_NOSIGPIPE).
- `writeFrameKind:payload:` — uses `writev(3)` with iovec of [4-byte BE length, kind byte, payload] on a dedicated writer queue (`dispatch_queue_create` SERIAL).
- Reader loop on a separate queue: `read()` into rolling NSMutableData, loop `DSFramerDecodeOne` until exhausted. Dispatch handlers as decoded.

```objc
// 0xC1 touch payload: [f64 BE x][f64 BE y][u8 phase][f64 BE pressure][u32 BE id] = 29 bytes
static void DSDecodeTouchPayload(NSData *p, double *x, double *y, uint8_t *phase, double *pressure, uint32_t *touchId) {
  const uint8_t *b = p.bytes;
  if (p.length < 29) return;
  uint64_t xN, yN, prN; uint32_t idN;
  memcpy(&xN, b+0, 8); memcpy(&yN, b+8, 8); *phase = b[16]; memcpy(&prN, b+17, 8); memcpy(&idN, b+25, 4);
  uint64_t xH = CFSwapInt64BigToHost(xN);
  uint64_t yH = CFSwapInt64BigToHost(yN);
  uint64_t prH = CFSwapInt64BigToHost(prN);
  uint32_t idH = CFSwapInt32BigToHost(idN);
  memcpy(x, &xH, 8); memcpy(y, &yH, 8); memcpy(pressure, &prH, 8);
  *touchId = idH;
}
```

```objc
#import <Foundation/Foundation.h>
#import <signal.h>
#import "Bridge.h"
#import "ScreenAttach.h"
#import "TouchInject.h"
#import "IpcServer.h"
#import "H264Encoder.h"

int main_probe(int argc, const char *argv[]);

static DSIpcServer *g_ipc = nil;
static DSH264Encoder *g_encoder = nil;

static void on_encoded(DSEncodedKind kind, const uint8_t *bytes, size_t len, void *userData) {
  if (!g_ipc) return;
  uint8_t kindByte = (uint8_t)kind; // 0x01 or 0x02
  [g_ipc writeFrameKind:kindByte payload:[NSData dataWithBytes:bytes length:len]];
}

static void on_frame(CVPixelBufferRef pb, double w, double h, void *userData) {
  if (!g_encoder) {
    g_encoder = [[DSH264Encoder alloc] initWithWidth:(int)w height:(int)h fps:30 bitrate:4000000
                                       outputCallback:on_encoded userData:NULL];
  }
  [g_encoder encodePixelBuffer:pb forceKeyframe:NO];
}

static void on_error(const char *msg, void *userData) {
  if (g_ipc) [g_ipc writeFrameKind:0xFF payload:[NSData dataWithBytes:msg length:strlen(msg)]];
}

static void on_touch(double x, double y, uint8_t phase, double pressure, uint32_t tid, void *u) {
  bridge_send_touch(x, y, phase, pressure, tid);
}

static void on_quit(void *u) {
  exit(0);
}

int main(int argc, const char *argv[]) {
  signal(SIGPIPE, SIG_IGN);
  @autoreleasepool {
    if (argc >= 2 && strcmp(argv[1], "--probe") == 0) {
      return main_probe(argc, argv);
    }
    // --udid <UDID> --socket <path>
    const char *udid = NULL, *sockPath = NULL;
    for (int i = 1; i + 1 < argc; i++) {
      if (strcmp(argv[i], "--udid") == 0) udid = argv[++i];
      else if (strcmp(argv[i], "--socket") == 0) sockPath = argv[++i];
    }
    if (!udid || !sockPath) {
      fprintf(stderr, "usage: %s --udid <UDID> --socket <path>\n  or:    %s --probe <UDID>\n", argv[0], argv[0]);
      return 64;
    }

    g_ipc = [[DSIpcServer alloc] initWithSocketPath:@(sockPath)];
    [g_ipc setTouchHandler:on_touch userData:NULL];
    [g_ipc setQuitHandler:on_quit userData:NULL];
    NSError *err = nil;
    if (![g_ipc startAndAcceptOneClientOrFail:&err]) {
      fprintf(stderr, "ipc start failed: %s\n", [err.description UTF8String]); return 2;
    }

    int rc = DSAttachToSimulator(udid, on_frame, on_error, NULL);
    if (rc != 0) { fprintf(stderr, "attach failed rc=%d\n", rc); return 3; }

    [[NSRunLoop currentRunLoop] run];
    return 0;
  }
}
```
