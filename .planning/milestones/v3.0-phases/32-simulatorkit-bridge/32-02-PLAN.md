#import <AppKit/AppKit.h>

// Forward-declare the private C symbol exported by SimulatorKit / CoreSimulator.
// (No public header ships; linker resolves via the framework dyld load.)
extern void *IndigoHIDMessageForMouseNSEvent(NSEvent *event, size_t *outLen, void (**outFreeFn)(void *));

// Lightweight wrapper class — holds the malloc'd void* + freeFn, sent to SimDeviceLegacyHIDClient.
@interface DSIndigoMessage : NSObject
@property (nonatomic, readonly) const void *bytes;
@property (nonatomic, readonly) NSUInteger length;
@end

@implementation DSIndigoMessage {
  void *_bytes;
  NSUInteger _length;
  void (*_freeFn)(void *);
}
- (instancetype)initWithBytes:(void *)bytes length:(NSUInteger)length freeFn:(void (*)(void *))freeFn {
  if ((self = [super init])) { _bytes = bytes; _length = length; _freeFn = freeFn; }
  return self;
}
- (const void *)bytes { return _bytes; }
- (NSUInteger)length { return _length; }
- (void)dealloc { if (_freeFn && _bytes) _freeFn(_bytes); }
@end

static NSEvent *DSSynthesizeNSEvent(double x_ratio, double y_ratio, uint8_t phase, double pressure, uint32_t touchId, CGSize displaySize) {
  // Mirror kittyfarm L944-948 — pass ratio directly, helper consumes ratio + displaySize separately.
  // NSEventType: 0=down → NSEventTypeLeftMouseDown, 1=move → Dragged, 2=up → Up
  NSEventType type;
  switch (phase) {
    case 0: type = NSEventTypeLeftMouseDown; break;
    case 1: type = NSEventTypeLeftMouseDragged; break;
    case 2: type = NSEventTypeLeftMouseUp; break;
    default: type = NSEventTypeLeftMouseUp; break;
  }
  NSPoint loc = NSMakePoint(x_ratio * displaySize.width, y_ratio * displaySize.height);
  NSEvent *ev = [NSEvent mouseEventWithType:type
                                    location:loc
                              modifierFlags:0
                                    timestamp:[[NSProcessInfo processInfo] systemUptime]
                                  windowNumber:0
                                      context:nil
                                  eventNumber:(NSInteger)touchId
                                  clickCount:1
                                    pressure:(float)pressure];
  return ev;
}

int bridge_send_touch(double x_ratio, double y_ratio, uint8_t phase, double pressure, uint32_t touchId) {
  id hidClient = DSCurrentHIDClient();
  if (!hidClient) { fprintf(stderr, "bridge_send_touch: not attached\n"); return -1; }

  CGSize sz = DSDisplayPixelSize();
  if (sz.width == 0 || sz.height == 0) { fprintf(stderr, "bridge_send_touch: display size unknown — frame callback hasn't fired\n"); return -2; }

  NSEvent *ev = DSSynthesizeNSEvent(x_ratio, y_ratio, phase, pressure, touchId, sz);
  size_t len = 0;
  void (*freeFn)(void *) = NULL;
  void *bytes = IndigoHIDMessageForMouseNSEvent(ev, &len, &freeFn);
  if (!bytes || len == 0) { fprintf(stderr, "IndigoHIDMessageForMouseNSEvent returned empty\n"); return -3; }

  DSIndigoMessage *msg = [[DSIndigoMessage alloc] initWithBytes:bytes length:len freeFn:freeFn];

  // Sync send (kittyfarm L895-911 — dispatch_semaphore-blocked).
  dispatch_queue_t completionQueue = dispatch_queue_create("com.devicestream.sim-private-hid-completion", DISPATCH_QUEUE_SERIAL);
  dispatch_semaphore_t sem = dispatch_semaphore_create(0);
  __block NSError *outErr = nil;

  SEL sendSel = NSSelectorFromString(@"sendWithMessage:freeWhenDone:completionQueue:completion:");
  NSMethodSignature *sig = [hidClient methodSignatureForSelector:sendSel];
  NSInvocation *inv = [NSInvocation invocationWithMethodSignature:sig];
  inv.target = hidClient;
  inv.selector = sendSel;
  [inv setArgument:&msg atIndex:2];
  BOOL freeWhenDone = YES;
  [inv setArgument:&freeWhenDone atIndex:3];
  [inv setArgument:&completionQueue atIndex:4];
  void (^completion)(NSError *) = ^(NSError *err) { outErr = err; dispatch_semaphore_signal(sem); };
  [inv setArgument:&completion atIndex:5];
  [inv invoke];

  if (dispatch_semaphore_wait(sem, dispatch_time(DISPATCH_TIME_NOW, 2 * NSEC_PER_SEC)) != 0) {
    fprintf(stderr, "HID send timed out\n"); return -4;
  }
  if (outErr) { fprintf(stderr, "HID send error: %s\n", [outErr.description UTF8String]); return -5; }
  return 0;
}
