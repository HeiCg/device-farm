```javascript
// Postinstall hook for @device-stream — runs build scripts that have to fire
// after `npm install`. Today: just the sim-capture-private build is gated
// behind a Wave-0 stub (no-op in Wave 0).
//
// Extended by:
//   - Plan 32-04 (T-32.6) — flip stub to actually build on macOS hosts
//   - Future phases as needed
'use strict';

const { spawnSync } = require('node:child_process');
const path = require('node:path');
const os = require('node:os');

function shouldBuild() {
  if (process.env.DEVICE_STREAM_SKIP_BUILD === '1') return false;
  if (process.platform !== 'darwin') return false; // sim-capture-private is macOS-only
  if (os.arch() !== 'arm64') return false;          // arm64-only per Risks in 32-RESEARCH.md
  return true;
}

if (!shouldBuild()) {
  console.log('[device-stream postinstall] skipping native build (platform=%s arch=%s)', process.platform, os.arch());
  process.exit(0);
}

const scriptDir = path.resolve(__dirname);

const tasks = [
  // Wave 0: gated by SIM_PRIVATE_WAVE0_STUB=1 default; Plan 32-04 will set it to 0.
  ['bash', [path.join(scriptDir, 'build-sim-capture-private.sh')], 'sim-capture-private'],
];

for (const [cmd, args, name] of tasks) {
  console.log('[device-stream postinstall] building %s', name);
  const r = spawnSync(cmd, args, { stdio: 'inherit', env: { ...process.env } });
  if (r.status !== 0) {
    console.warn('[device-stream postinstall] %s build failed (status=%d) — install continuing', name, r.status);
    // Do NOT block npm install on a stub failure. Plan 32-04 hardens this.
  }
}
```
