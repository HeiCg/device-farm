`DSLoadPrivateFrameworks()` body:
```objc
int DSLoadPrivateFrameworks(void) {
    static int s_state = 0; // 0 unloaded, 1 ok, -1 failed
    static dispatch_once_t once;
    dispatch_once(&once, ^{
        void *cs = dlopen("/Library/Developer/PrivateFrameworks/CoreSimulator.framework/CoreSimulator", RTLD_LAZY);
        if (!cs) { fprintf(stderr, "dlopen CoreSimulator failed: %s\n", dlerror()); s_state = -1; return; }
        // Resolve $DEVELOPER_DIR. Fallback to /Applications/Xcode.app/Contents/Developer.
        const char *devDir = getenv("DEVELOPER_DIR");
        if (!devDir || !*devDir) devDir = "/Applications/Xcode.app/Contents/Developer";
        char skPath[1024];
        snprintf(skPath, sizeof(skPath), "%s/Library/PrivateFrameworks/SimulatorKit.framework/SimulatorKit", devDir);
        void *sk = dlopen(skPath, RTLD_LAZY);
        if (!sk) { fprintf(stderr, "dlopen SimulatorKit failed: %s\n", dlerror()); s_state = -1; return; }
        s_state = 1;
    });
    return s_state == 1 ? 0 : -1;
}
```

const DSSymbolSpec kDSCriticalSymbols[8] = {
  // Swift mangled symbols (trie-resolved)
  {.prefix="$sSo9SimDeviceC12SimulatorKitE13screenAdapter",        .suffix="vg",  .role="SimDevice.screenAdapter.getter"},
  {.prefix="$s12SimulatorKit22SimDeviceScreenAdapterC7screens",     .suffix="vg",  .role="SimDeviceScreenAdapter.screens.getter"},

  // Obj-C selector probes
  {.classOrSelector="SimServiceContext",         .selectorIfClass="contextForDeveloperDir:connectionType:error:", .role="SimServiceContext +contextForDeveloperDir:"},
  {.classOrSelector="SimDeviceLegacyHIDClient", .selectorIfClass="initWithDevice:error:",                          .role="SimDeviceLegacyHIDClient -initWithDevice:"},
  {.classOrSelector="SimDeviceScreen",          .selectorIfClass="initWithDevice:screenID:",                       .role="SimDeviceScreen -initWithDevice:screenID:"},
  {.classOrSelector="SimDeviceScreen",          .selectorIfClass="registerScreenCallbacksWithUUID:callbackQueue:frameCallback:surfacesChangedCallback:propertiesChangedCallback:", .role="SimDeviceScreen -registerScreenCallbacks..."},
  {.classOrSelector="SimDeviceLegacyHIDClient", .selectorIfClass="sendWithMessage:freeWhenDone:completionQueue:completion:", .role="SimDeviceLegacyHIDClient -sendWithMessage:"},
};
const int kDSCriticalSymbolCount = 8;

int DSProbeCriticalSymbols(void) {
  if (DSLoadPrivateFrameworks() != 0) return 0;
  int resolved = 0;
  for (int i = 0; i < kDSCriticalSymbolCount; i++) {
    const DSSymbolSpec *s = &kDSCriticalSymbols[i];
    BOOL ok = NO;
    if (s->prefix && s->suffix) {
      ok = DSResolveSwiftSymbol(s->prefix, s->suffix, s->role) != NULL;
    } else if (s->classOrSelector && s->selectorIfClass) {
      Class c = NSClassFromString([NSString stringWithUTF8String:s->classOrSelector]);
      SEL sel = NSSelectorFromString([NSString stringWithUTF8String:s->selectorIfClass]);
      ok = (c != Nil) && ([c respondsToSelector:sel] || [c instancesRespondToSelector:sel]);
    } else if (s->classOrSelector) {
      ok = NSClassFromString([NSString stringWithUTF8String:s->classOrSelector]) != Nil;
    }
    if (ok) resolved++;
    else fprintf(stderr, "MISSING: %s\n", s->role);
  }
  return resolved;
}

int bridge_attach(const char *udid) {

  fprintf(stderr, "bridge_attach: not implemented yet (Plan 32-02)\n");
  return -1;
}
