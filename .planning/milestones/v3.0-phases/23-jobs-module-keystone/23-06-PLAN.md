    /**
     * Phase 23 Plan 23-06 — lifecycle-ownership.spec [SC1 grep-guards].
     *
     * Proves the post-Plan-23-04 codebase has zero anti-patterns:
     *   - .catch(() => {})              — error swallowing (use job.failed emit instead)
     *   - setTimeout(...broadcaster.cleanup) — DEFERRED-22-D legacy (use job.cleanup.requested instead)
     *   - import from '../streaming/internal/'  — DEFERRED-22-F legacy (use fastify.streamingModule decorator surface)
     *   - bus.emit() outside events.ts  — direct emit (use makeJobsEmitters helpers)
     *
     * Scoped narrowly per RESEARCH §Pitfall 7 — checks only jobs/internal/* + jobs/job-service.ts.
     * DB-FREE.
     */
    import { describe, it, expect } from 'vitest';
    import { readFileSync } from 'node:fs';
    import { join } from 'node:path';

    const FILES = [
      'server/jobs/internal/executor.ts',
      'server/jobs/internal/module.ts',
      'server/jobs/internal/subscribers.ts',
      'server/jobs/job-service.ts',
    ];

    function loadFile(rel: string): string {
      return readFileSync(join(process.cwd(), rel), 'utf-8');
    }

    describe('jobs lifecycle ownership [SC1 grep-guards]', () => {
      it('zero `.catch(() => {})` swallow patterns in executor + module + subscribers + shim', () => {
        const re = /\.catch\(\(\) => \{\}\)/g;

          const src = loadFile(f);

        }

        const re = /setTimeout\([^)]*broadcaster[^)]*cleanup/gs;
        for (const f of FILES) {
          const src = loadFile(f);
          const matches = src.match(re) ?? [];
          expect(matches.length, `${f} has setTimeout-broadcaster-cleanup at ${matches.length} sites`).toBe(0);
        }
      });

      it('zero imports from "../streaming/internal/" (DEFERRED-22-F resolved)', () => {
        const re = /from ['"]\.\.?\/streaming\/internal\//g;
        for (const f of FILES) {
          const src = loadFile(f);
          const matches = src.match(re) ?? [];
          expect(matches.length, `${f} has streaming/internal imports at ${matches.length} sites`).toBe(0);
        }
      });

      it('zero direct `bus.emit(` outside events.ts (use makeJobsEmitters helpers)', () => {
        const re = /\bbus\.emit\(/g;
        // events.ts is allowlisted (createEventHelpers internally calls bus.emit)

          const src = loadFile(f);
          const matches = src.match(re) ?? [];
          expect(matches.length, `${f} has direct bus.emit() at ${matches.length} sites`).toBe(0);
        }
      });

      it('back-compat shim (job-service.ts) is small (≤ 80 lines)', () => {

      });
    });
