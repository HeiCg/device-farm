    /**
     * Phase 23 Plan 23-05 — Jobs admin routes (drain procedure).
     *
     * RESOLVES PITFALL 1: pg-boss v12 has NO `paused` flag on updateQueue.
     * Drain mechanism: boss.offWork(name, {wait:false}) stops THIS server's
     * worker(s); in-flight workers complete; admission check on enqueueJob
     * blocks new work. system_state row persists across server restarts so
     * drain state survives restart (plugin onReady reads it).
     *
     * Auth: any-valid-key gate via fastify.authService.validateKey.
     * DEFERRED-23-A: Phase 26 Auth Module formalizes admin claim + requireAdmin
     * middleware. TODO comment in route handler.
     *
     * RESEARCH §Code Examples §3 + §Pattern 6.
     */
    import { eq } from 'drizzle-orm';
    import { z } from 'zod';
    import type { FastifyInstance } from 'fastify';
    import type pino from 'pino';
    import * as schema from '../../db/schema.js';
    import { QUEUE_NAMES } from '../../queue/names.js';
    import { JOB_EXECUTE_QUEUE_NAME } from '../queue.js';
    import type { JobsModule } from './module.js';

    const drainQuerySchema = z.object({
      timeout: z.coerce.number().int().min(1).max(1800).default(300),
    });

    const drainResponseDrainedSchema = z.object({
      drained: z.literal(true),
      in_flight: z.literal(0),
      drained_at: z.string().datetime(),
    });

    const drainResponseTimeoutSchema = z.object({
      drained: z.literal(false),
      in_flight: z.number().int().nonnegative(),
      timeout: z.literal(true),
    });

    const drainResponseSchema = z.discriminatedUnion('drained', [
      drainResponseDrainedSchema,
      drainResponseTimeoutSchema,
    ]).meta({ id: 'DrainResponse' });

    const resumeResponseSchema = z.object({ resumed: z.literal(true) }).meta({ id: 'DrainResumeResponse' });

    export interface RegisterJobsAdminRoutesDeps {
      fastify: FastifyInstance;
      jobsModule: JobsModule;
      logger: pino.Logger;
    }

    export async function registerJobsAdminRoutes(deps: RegisterJobsAdminRoutesDeps): Promise<void> {
      const { fastify, jobsModule, logger } = deps;
      const log = logger.child({ module: 'jobs.routes' });

      fastify.post(
        '/admin/drain',
        {
          // TODO(Phase 26 — DEFERRED-23-A): replace any-valid-key gate with requireAdmin claim
          preHandler: fastify.authService?.validateKey,
          schema: {
            querystring: drainQuerySchema,
            response: { 200: drainResponseSchema },
          },
        },
        async (req, reply) => {
          const start = Date.now();
          const timeoutMs = (req.query as z.infer<typeof drainQuerySchema>).timeout * 1000;

          // 1. Persist drain intent (idempotent upsert) — restart-safe.
          await fastify.db
            .insert(schema.systemState)
            .values({
              key: 'drain_requested_at',
              value: { iso: new Date().toISOString() },
            })
            .onConflictDoUpdate({
              target: schema.systemState.key,
              set: { value: { iso: new Date().toISOString() }, updatedAt: new Date() },
            });

          // 2. Stop fetchers — workers complete current handler, refuse new.
          // wait:false so we own the polling loop; offWork is per-queue (removes all workers for that name on this PgBoss instance).
          await fastify.boss.offWork(JOB_EXECUTE_QUEUE_NAME, { wait: false }).catch((err) => {
            log.warn({ err }, 'offWork(JOB_EXECUTE) errored — likely already stopped');
          });
          await fastify.boss.offWork(QUEUE_NAMES.RECORDING_UPLOAD, { wait: false }).catch((err) => {
            log.warn({ err }, 'offWork(RECORDING_UPLOAD) errored — likely already stopped');
          });

          // 3. Long-poll until in-flight = 0 OR timeout
          while (Date.now() - start < timeoutMs) {
            const inFlight = jobsModule.getInFlightCount();
            if (inFlight === 0) {
              const drainedAt = new Date().toISOString();
              jobsModule.emit.drainCompleted(drainedAt, {
                drainedAt,
                durationMs: Date.now() - start,
              });
              return reply.send({ drained: true as const, in_flight: 0 as const, drained_at: drainedAt });
            }
            await new Promise((resolve) => setTimeout(resolve, 1000));
          }

          // 4. Timeout — return current count
          return reply.send({
            drained: false as const,
            in_flight: jobsModule.getInFlightCount(),
            timeout: true as const,
          });
        },
      );

      fastify.post(
        '/admin/drain/resume',
        {
          preHandler: fastify.authService?.validateKey,
          schema: { response: { 200: resumeResponseSchema } },
        },
        async (_req, reply) => {
          // 1. Clear DB flag
          await fastify.db.delete(schema.systemState).where(eq(schema.systemState.key, 'drain_requested_at'));

          // 2. Re-register workers (offWork removed them — this brings them back).
          await jobsModule.registerWorkerAndSubscribers();
          // Note: artifactsModule's recording.upload worker also needs re-register.
          // The artifacts module factory exposes registerWorkersAndSubscribers; call if available.
          if ('artifactsModule' in fastify) {
            const am = (fastify as { artifactsModule?: { registerWorkersAndSubscribers?: () => Promise<void> } }).artifactsModule;
            await am?.registerWorkersAndSubscribers?.();
          }

          // 3. Emit + ack
          const resumedAt = new Date().toISOString();
          jobsModule.emit.drainResumed(resumedAt, { resumedAt });

          return reply.send({ resumed: true as const });
        },
      );
    }

    /**
     * Restart safety helper — called by createJobsModule.registerWorkerAndSubscribers
     * during plugin boot. If system_state.drain_requested_at exists, immediately
     * offWork the worker before it processes any queue items.
     */
    export async function honorDrainOnBoot(fastify: FastifyInstance, log: pino.Logger): Promise<void> {
      const rows = await fastify.db
        .select()
        .from(schema.systemState)
        .where(eq(schema.systemState.key, 'drain_requested_at'))
        .limit(1);
      if (rows.length > 0) {
        log.warn({ rows }, 'Boot detected drain_requested_at — calling offWork immediately');
        await fastify.boss.offWork(JOB_EXECUTE_QUEUE_NAME, { wait: false }).catch(() => undefined);
        await fastify.boss.offWork(QUEUE_NAMES.RECORDING_UPLOAD, { wait: false }).catch(() => undefined);
      }
    }
    ```

    **NOTE on offWork(name) targeting**: pg-boss v12 `offWork(idOrName)` accepts either a worker id (returned by `boss.work()`) OR a queue name. By name, it removes ALL workers for that queue from this PgBoss instance. Verify by reading `node_modules/pg-boss/dist/manager.d.ts:47` to confirm signature. If the API requires the worker id specifically, switch to passing `workerIds` from createJobsModule (the factory tracks them).

    # Runbook: Graceful Drain

    Phase 23 Plan 23-05 — drain procedure for safe operator-initiated quiescence.

    ## Why drain?

    The Device Farm runs Maestro test execution against Android emulators. Killing the server mid-execution leaves emulators in inconsistent states (allocated but never released; recordings half-written). Drain stops new work and waits for in-flight to finish naturally.

    ## Mechanism

    Drain is a 3-step state change:

    1. **Persistent flag:** A row in `system_state` (`key = 'drain_requested_at'`) records the intent. This survives server restart — the plugin reads it on boot via `honorDrainOnBoot` and immediately calls `boss.offWork`.
    2. **Worker stop:** `boss.offWork(JOB_EXECUTE_QUEUE_NAME, {wait:false})` removes the worker from this server's pg-boss instance. In-flight workers complete their current handler. New jobs in the queue are NOT picked up (single-node project — no other server can pick them up either).
    3. **Admission gate:** `fastify.jobsModule.enqueueJob` reads the system_state row before `queue.send`; presence → throws 503 `system_draining`.

    ## Procedure

    ### Drain

    ```bash
    curl -X POST 'http://localhost:3000/admin/drain?timeout=300' \
      -H 'X-API-Key: <YOUR_KEY>'
    ```

    Response (success):
    ```json
    {"drained": true, "in_flight": 0, "drained_at": "2026-05-08T12:00:00.000Z"}
    ```

    Response (timeout — in-flight stuck):
    ```json
    {"drained": false, "in_flight": 2, "timeout": true}
    ```

    ### Restart server

    Once `drained:true`:
    ```bash
    sudo systemctl restart device-farm   # or your process manager
    ```

    On boot, plugin reads `system_state.drain_requested_at` row, calls `offWork` immediately. Worker stays parked.

    ### Resume

    ```bash
    curl -X POST 'http://localhost:3000/admin/drain/resume' \
      -H 'X-API-Key: <YOUR_KEY>'
    ```

    Response:
    ```json
    {"resumed": true}
    ```

    The plugin re-registers the worker; `enqueueJob` accepts new requests.

    ## Failure modes

    - **Timeout — in-flight stuck:** A job has hung (zombie qemu, network wait, etc.). Manual recovery: identify via `fastify.jobsModule.runningJobs` keys (logs); `kill -9` the qemu process; the worker's AbortController completes; in-flight count decrements. Alternatively, force-restart server while drain row present — plugin honors drain on next boot.
    - **Resume fails:** if `registerWorkerAndSubscribers` errors during resume, the system_state row is already deleted but the worker is not registered. Fix: restart server (plugin registers worker via createJobsModule on boot; with no drain row, work proceeds normally).
    - **Database unreachable:** Drain endpoint returns 5xx. The system_state row may not have been written. Manual recovery: when DB recovers, POST /admin/drain again.

    ## Auth

    Both endpoints require any-valid API key (any key created via `auth.key.created` event). Phase 26 Auth Module formalizes admin claim + `requireAdmin` middleware (DEFERRED-23-A).

    ## Events

    `system.drain.completed` event is persisted to the `events` table on successful drain. Audit trail — query by `event_type` for drain rehearsal history.

    `system.drain.resumed` event is persisted on resume.

    ## Observability

    Watch logs: `docker logs device-farm 2>&1 | grep -i drain` shows drain-related lines including the warn line from `honorDrainOnBoot` if the server boots into a drained state.
    ```
