    ```typescript
    /**
     * Phase 19 / Plan 19-04 — Reporting queue DLQ pipeline spec (ROADMAP SC1).
     *
     * Proves: 5× 500 Internal Server Error → after pg-boss exhausts retries +
     * runs maintenance → the DLQ queue contains the failed payload; the Fastify
     * app does NOT crash during any of the 5 attempts.
     *
     * DB-gated (TEST_DATABASE_URL); runs in ~8-15s with maintenanceIntervalSeconds:1.
     */
    import { afterAll, beforeAll, describe, it, expect, vi } from 'vitest';
    import Fastify, { type FastifyInstance } from 'fastify';
    import fp from 'fastify-plugin';
    import { createServer, type Server } from 'node:http';
    import postgres from 'postgres';

    import reportingPlugin from '../plugin.js';
    import queuePlugin from '../../queue/plugin.js';
    import busPlugin from '../../bus/plugin.js';
    import dbPlugin from '../../db/plugin.js';
    import correlationPlugin from '../../correlation/plugin.js';
    import { WEBHOOK_DELIVER_QUEUE_NAME, WEBHOOK_DELIVER_DLQ_QUEUE_NAME } from '../queue.js';

    const TEST_DATABASE_URL = process.env.TEST_DATABASE_URL ?? process.env.DATABASE_URL;
    const HAS_DB = typeof TEST_DATABASE_URL === 'string' && TEST_DATABASE_URL.length > 0;
    const SCHEMA = 'pgboss_reporting_queue_spec';

    describe.skipIf(!HAS_DB)('reporting queue DLQ pipeline (Phase 19-04 — SC1)', () => {
      let app: FastifyInstance;
      let client: ReturnType<typeof postgres>;
      let failingServer: Server;
      let failingUrl: string;
      let requestCount = 0;

      beforeAll(async () => {
        client = postgres(TEST_DATABASE_URL!);
        await client.unsafe(`DROP SCHEMA IF EXISTS ${SCHEMA} CASCADE`);

        // Local HTTP server that ALWAYS returns 500 — triggers retry-exhaustion.
        failingServer = createServer((req, res) => {
          requestCount++;
          let body = '';
          req.on('data', (c) => { body += c.toString(); });
          req.on('end', () => {
            res.writeHead(500, { 'Content-Type': 'text/plain' });
            res.end('nope');
          });
        });
        failingUrl = await new Promise((resolve) => {
          failingServer.listen(0, () => {
            const addr = failingServer.address();
            if (addr && typeof addr === 'object') resolve(`http://127.0.0.1:${addr.port}/hook`);
            else throw new Error('Failed to bind failingServer');
          });
        });

        // Stub config plugin that points webhooks.url at the failing server.
        const stubConfigPlugin = fp(async (fastify: FastifyInstance) => {
          fastify.decorate('config', {
            database_url: TEST_DATABASE_URL!,
            webhooks: { url: failingUrl, secret: undefined, timeout_ms: 2_000, max_retries: 5 },
          } as never);
        }, { name: 'config' });

        app = Fastify({ logger: false });
        await app.register(stubConfigPlugin);
        await app.register(correlationPlugin);
        await app.register(dbPlugin);
        await app.register(busPlugin);
        await app.register(queuePlugin, { schema: SCHEMA, maintenanceIntervalSeconds: 1 });
        await app.register(reportingPlugin);
        await app.ready();
      }, 30_000);

      afterAll(async () => {
        await app?.close();
        await new Promise<void>((r) => failingServer?.close(() => r()));
        await client?.unsafe(`DROP SCHEMA IF EXISTS ${SCHEMA} CASCADE`).catch(() => {});
        await client?.end();
      }, 30_000);

      it('5× 500 → DLQ row appears; server does not crash', async () => {
        const payload = { event: 'job.completed', job: { id: 'test-job-sc1' } };
        const requestCountBefore = requestCount;

        // Enqueue directly (bypassing bus subscriber for deterministic test scope).
        const jobId = await app.queue.send(WEBHOOK_DELIVER_QUEUE_NAME, {
          url: failingUrl,
          payload,
        }, {});
        expect(jobId).not.toBeNull();

        // Wait for pg-boss to exhaust retries AND move to DLQ (maintenanceIntervalSeconds:1 means ~1s lag after final failure).
        // With retryDelay:1 + retryBackoff:true, 5 retries take ~1+2+4+8+16 = 31s worst case — we use retryDelayMax:30 cap,
        // plus 1s maintenance tick. Budget: 40s. We waitFor with 45s timeout and 1s poll.
        await vi.waitFor(async () => {
          const dlqJobs = await app.boss.findJobs(WEBHOOK_DELIVER_DLQ_QUEUE_NAME);
          expect(dlqJobs.length).toBeGreaterThanOrEqual(1);
        }, { timeout: 45_000, interval: 1_000 });

        // Assert: at least 5 requests hit the failing server (retries fired).
        // pg-boss may have fired MORE than 5 if the test harness retries; assert >=5.
        expect(requestCount - requestCountBefore).toBeGreaterThanOrEqual(5);

        // Assert: the DLQ row payload matches our enqueue.
        const dlqJobs = await app.boss.findJobs(WEBHOOK_DELIVER_DLQ_QUEUE_NAME);
        expect(dlqJobs.length).toBeGreaterThanOrEqual(1);
        const matching = dlqJobs.filter((j) => {
          const data = j.data as { payload?: { payload?: unknown }; url?: string };
          return data?.url === failingUrl || (data?.payload as { url?: string } | undefined)?.url === failingUrl;
        });
        expect(matching.length).toBeGreaterThanOrEqual(1);

        // Assert: app is still responsive — health-check inject must succeed.
        const health = await app.inject({ method: 'GET', url: '/health' }).catch(() => null);
        // /health may not exist in this test harness; if present, must be 200.
        if (health) expect(health.statusCode).toBe(200);
      }, 60_000);  // per-test timeout
    });
    ```

The main SC1 invariant is that `app.boss.findJobs` successfully returns and the app responds to ANY inject without throwing. The optional health-check line is defensive; remove if the harness genuinely doesn't have `/health`.
