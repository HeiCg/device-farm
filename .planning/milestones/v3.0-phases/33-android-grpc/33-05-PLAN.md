```bash
#!/usr/bin/env bash
# Phase 33 — build the android-grpc-stream Go daemon (production version).
# Replaces the Wave 0 stub that exited non-zero with "not implemented".
set -euo pipefail

SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
MODULE_DIR="$SCRIPT_DIR/../native-servers/android-grpc"
BIN_DIR="$SCRIPT_DIR/../bin"

# Preflight: protoc + plugins
for tool in protoc protoc-gen-go protoc-gen-go-grpc; do
  if ! command -v "$tool" >/dev/null 2>&1; then
    echo "ERROR: $tool not found." >&2
    case "$tool" in
      protoc) echo "  Install: brew install protobuf" >&2 ;;
      protoc-gen-go) echo "  Install: go install google.golang.org/protobuf/cmd/protoc-gen-go@latest" >&2 ;;
      protoc-gen-go-grpc) echo "  Install: go install google.golang.org/grpc/cmd/protoc-gen-go-grpc@latest" >&2 ;;
    esac
    exit 1
  fi
done

# Build
cd "$MODULE_DIR"
echo "[build-android-grpc] make proto..."
make proto
echo "[build-android-grpc] make build..."
make build

# Stage
mkdir -p "$BIN_DIR"
cp -f "$MODULE_DIR/bin/android-grpc-stream" "$BIN_DIR/android-grpc-stream"
chmod +x "$BIN_DIR/android-grpc-stream"
echo "[build-android-grpc] staged at $BIN_DIR/android-grpc-stream"
"$BIN_DIR/android-grpc-stream" --help >/dev/null
echo "[build-android-grpc] OK"
```

3. Write `device-stream/scripts/smoke-android-grpc.sh` (~120 LOC). Mirror the structure of `smoke-sim-private.sh`. Key sections:
```bash
#!/usr/bin/env bash
set -euo pipefail
AVD="${1:?usage: smoke-android-grpc.sh <avd>}"
GRPC_PORT="${GRPC_PORT:-8554}"
SOCK="/tmp/device-stream-android-emu-$(whoami)-smoke.sock"
BIN="$(cd "$(dirname "${BASH_SOURCE[0]}")/.." && pwd)/bin/android-grpc-stream"

[ -x "$BIN" ] || { echo "binary missing at $BIN — run build-android-grpc.sh"; exit 1; }

# Boot emulator headless
emulator -avd "$AVD" -grpc "$GRPC_PORT" -no-window -no-audio -gpu swiftshader_indirect &
EMU_PID=$!
trap "kill $EMU_PID 2>/dev/null || true; rm -f $SOCK" EXIT

# Wait for boot
for i in $(seq 1 60); do
  if adb -e shell getprop sys.boot_completed 2>/dev/null | tr -d '\r' | grep -q '^1$'; then
    break
  fi
  sleep 2
done

# Probe
"$BIN" --probe --serial emulator-5554 --grpc-port "$GRPC_PORT" || { echo "probe failed"; exit 1; }

# Spawn daemon
"$BIN" --serial emulator-5554 --grpc-port "$GRPC_PORT" --socket "$SOCK" &
DAEMON_PID=$!
trap "kill $DAEMON_PID 2>/dev/null || true; kill $EMU_PID 2>/dev/null || true; rm -f $SOCK" EXIT

# Read 30 frames via python3 length-prefix framer
python3 - <<'EOF'
import socket, struct, sys, time
sock = "%s"
s = socket.socket(socket.AF_UNIX, socket.SOCK_STREAM)
for _ in range(50):
  try: s.connect(sock); break
  except OSError: time.sleep(0.1)
frames = 0
start = time.time()
while frames < 30 and time.time() - start < 30:
  hdr = s.recv(4)
  if len(hdr) < 4: continue
  length = struct.unpack('>I', hdr)[0]
  payload = b''
  while len(payload) < length:
    chunk = s.recv(length - len(payload))
    if not chunk: break
    payload += chunk
  kind = payload[0]
  if kind == 0x02: frames += 1
  if kind == 0x01: print("paramSets received", file=sys.stderr)
# Send 0xC9 quit
s.send(struct.pack('>I', 1) + bytes([0xC9]))
s.close()
print(f"received {frames} H.264 access units")
assert frames >= 30, f"only {frames} frames in 30s"
EOF

echo "smoke-android-grpc: OK"
```
(The Python multiline string injection of `$SOCK` requires `%s % sock` or sed substitution — keep the helper inline; fix with a heredoc that expands variables BEFORE the python EOF block.)

```bash
#!/usr/bin/env bash
set -euo pipefail
AVD="${1:?usage: android-grpc-soak.sh <avd> [--duration 30m]}"
DUR_MIN=30
# parse --duration ...

# ... boot emulator, spawn daemon ...
# Sample RSS every 30s, gate growth ≤ 50MB
SAMPLES=0
INITIAL_RSS=$(ps -o rss= -p "$DAEMON_PID" | tr -d ' ')
while [ "$SAMPLES" -lt $((DUR_MIN * 2)) ]; do
  sleep 30
  CUR_RSS=$(ps -o rss= -p "$DAEMON_PID" | tr -d ' ')
  GROWTH_KB=$((CUR_RSS - INITIAL_RSS))
  echo "sample $SAMPLES: rss=${CUR_RSS}kb growth=${GROWTH_KB}kb"
  if [ "$GROWTH_KB" -gt 51200 ]; then
    echo "ERROR: RSS growth ${GROWTH_KB}kb > 50MB ceiling"
    exit 1
  fi
  SAMPLES=$((SAMPLES + 1))
done
echo "soak: OK — RSS stable over ${DUR_MIN}min"
```
