```typescript
/**
 * Node-side client for the android-grpc-stream daemon (Phase 33, Android gRPC EmulatorController bridge).
 *
 * Wire format (mirrors device-stream/native-servers/android-grpc/ipc/framer.go):
 *
 *   Frame on the wire: [u32 BE length][u8 kind][payload]
 *     where `length` = 1 + payload.length (covers kind byte + payload bytes).
 *
 * Server -> client kinds:
 *   0x01 = SPS+PPS avcC paramSet blob
 *   0x02 = H.264 access unit (AVCC NAL bytes)
 *   0x03 = metadata {u32 BE width, u32 BE height, u32 BE fps} (12 bytes)
 *   0x10 = ack (ignored)
 *   0xFF = error (UTF-8 message)
 *
 * Client -> server kinds:
 *   0xC1 = touch (29 bytes): [f64 BE x_ratio][f64 BE y_ratio][u8 phase]
 *                            [f64 BE pressure][u32 BE touchId]
 *   0xC2 = key (9 bytes):    [u8 eventType (0=down,1=up)][u32 BE keyCode][u32 BE modMask]
 *   0xC9 = quit (empty)
 *
 * Lifecycle:
 *   - `GrpcEmuClient.spawn(serial, grpcPort, ws)` launches the daemon
 *     (`android-grpc-stream --serial <s> --grpc-port <p> --socket <path>`), waits
 *     for the Unix socket to appear, opens a SOCK_STREAM client, and returns a
 *     ready instance. Forwards inbound frames to the provided WebSocket.
 *   - Falls back to ScrcpyService when this client cannot be spawned — the
 *     fallback wiring lives in service.ts. This file is unconditional Node-side IPC.
 *
 * Env opt-out: DEVICE_STREAM_ANDROID_GRPC=0 — gate is enforced in service.ts.
 */
import { EventEmitter } from 'events';
import * as net from 'net';
import * as path from 'path';
import * as fs from 'fs';
import { ChildProcess, spawn } from 'child_process';
import type { WebSocket } from 'ws';

const DEFAULT_BINARY_PATH = path.resolve(__dirname, '../../../bin/android-grpc-stream');

export interface GrpcEmuClientOptions {
  binaryPath?: string;
  socketPath?: string;
  spawnTimeoutMs?: number;
}

export interface GrpcEmuTouch {
  xRatio: number; yRatio: number;
  phase: 0 | 1 | 2 | 3; // began/moved/ended/cancelled
  pressure: number;
  id: number;
}

export class GrpcEmuClient extends EventEmitter {
  private _stopped = false;
  private _firstFrameSeen = false;
  private _cachedParamSets: Buffer | null = null;
  private _readBuf = Buffer.alloc(0);

  private constructor(
    private readonly _serial: string,
    private readonly _grpcPort: number,
    private readonly _proc: ChildProcess,
    private readonly _sock: net.Socket,
    private readonly _ws: WebSocket | null,
  ) {
    super();
    this._attachParseLoop();
    this._attachProcessGuards();
  }

  get serial(): string { return this._serial; }
  get grpcPort(): number { return this._grpcPort; }

  static async spawn(
    serial: string,
    grpcPort: number,
    ws: WebSocket | null,
    opts?: GrpcEmuClientOptions,
  ): Promise<GrpcEmuClient> {
    const binaryPath = opts?.binaryPath ?? DEFAULT_BINARY_PATH;
    const socketPath = opts?.socketPath ?? `/tmp/device-stream-android-emu-${serial}.sock`;
    const timeoutMs = opts?.spawnTimeoutMs ?? 5000;

    if (!fs.existsSync(binaryPath)) {
      throw new Error(`GrpcEmuClient: binary missing at ${binaryPath} (DEVICE_STREAM_ANDROID_GRPC build skipped or postinstall failed)`);
    }
    if (fs.existsSync(socketPath)) fs.unlinkSync(socketPath);

    const proc = spawn(binaryPath, ['--serial', serial, '--grpc-port', String(grpcPort), '--socket', socketPath], {
      stdio: ['ignore', 'pipe', 'pipe'],
    });

    let exitedEarly = false;
    proc.once('exit', (code) => { if (!proc.killed && code !== 0) exitedEarly = true; });

    const deadline = Date.now() + timeoutMs;
    while (!fs.existsSync(socketPath)) {
      if (exitedEarly) throw new Error(`GrpcEmuClient: daemon exited before first frame (exit code observed)`);
      if (Date.now() > deadline) { proc.kill('SIGKILL'); throw new Error(`GrpcEmuClient: socket ${socketPath} not appeared within ${timeoutMs}ms`); }
      await new Promise((r) => setTimeout(r, 50));
    }

    const sock = net.createConnection(socketPath);
    await new Promise<void>((resolve, reject) => {
      sock.once('connect', () => resolve());
      sock.once('error', reject);
    });
    return new GrpcEmuClient(serial, grpcPort, proc, sock, ws);
  }

  private _attachParseLoop(): void {
    this._sock.on('data', (chunk: Buffer) => {
      this._readBuf = Buffer.concat([this._readBuf, chunk]);
      while (this._readBuf.length >= 5) {
        const length = this._readBuf.readUInt32BE(0);
        if (this._readBuf.length < 4 + length) break;
        const kind = this._readBuf.readUInt8(4);
        const payload = this._readBuf.subarray(5, 4 + length);
        this._readBuf = this._readBuf.subarray(4 + length);
        this._handleFrame(kind, payload);
      }
    });
    this._sock.on('error', (err) => this.emit('error', err));
    this._sock.on('close', () => { if (!this._stopped) this.emit('close'); });
  }

  private _handleFrame(kind: number, payload: Buffer): void {
    this._firstFrameSeen = true;
    switch (kind) {
      case 0x01: // SPS+PPS
        this._cachedParamSets = Buffer.from(payload);
        this._wsSend({ type: 'paramSets', data: payload.toString('base64') });
        this.emit('paramSets', payload);
        break;
      case 0x02: // H.264 AU
        this._wsSend({ type: 'frame', data: payload.toString('base64') });
        this.emit('frame', payload);
        break;
      case 0x03: // metadata {width, height, fps}
        if (payload.length === 12) {
          const width = payload.readUInt32BE(0);
          const height = payload.readUInt32BE(4);
          const fps = payload.readUInt32BE(8);
          this._wsSend({ type: 'metadata', width, height, fps, codec: 'h264' });
          this.emit('metadata', { width, height, fps });
        }
        break;
      case 0x10: break; // ack
      case 0xFF: this.emit('error', new Error(`daemon: ${payload.toString('utf8')}`)); break;
      default: /* ignore unknown */ break;
    }
  }

  private _wsSend(envelope: object): void {
    if (this._ws == null || (this._ws as any).readyState !== 1 /* OPEN */) return;
    try { (this._ws as any).send(JSON.stringify(envelope)); } catch { /* swallow */ }
  }

  // Replay cached param sets to a late-joining WS client.
  replayParamSets(ws: WebSocket): void {
    if (this._cachedParamSets) {
      try { (ws as any).send(JSON.stringify({ type: 'paramSets', data: this._cachedParamSets.toString('base64') })); } catch {}
    }
  }

  sendTouch(t: GrpcEmuTouch): void {
    const payload = Buffer.alloc(29);
    payload.writeDoubleBE(t.xRatio, 0);
    payload.writeDoubleBE(t.yRatio, 8);
    payload.writeUInt8(t.phase, 16);
    payload.writeDoubleBE(t.pressure, 17);
    payload.writeUInt32BE(t.id, 25);
    this._writeFrame(0xC1, payload);
  }

  sendTouchAtPixel(x: number, y: number, displayW: number, displayH: number, opts?: { phase?: 0|1|2|3; pressure?: number; id?: number }): void {
    const xRatio = x / Math.max(1, displayW - 1);
    const yRatio = y / Math.max(1, displayH - 1);
    this.sendTouch({ xRatio, yRatio, phase: opts?.phase ?? 0, pressure: opts?.pressure ?? 1, id: opts?.id ?? 0 });
  }

  sendKey(eventType: 'down' | 'up', keyCode: number, modMask = 0): void {
    const payload = Buffer.alloc(9);
    payload.writeUInt8(eventType === 'down' ? 0 : 1, 0);
    payload.writeUInt32BE(keyCode, 1);
    payload.writeUInt32BE(modMask, 5);
    this._writeFrame(0xC2, payload);
  }

  private _writeFrame(kind: number, payload: Buffer): void {
    const frame = Buffer.alloc(4 + 1 + payload.length);
    frame.writeUInt32BE(1 + payload.length, 0);
    frame.writeUInt8(kind, 4);
    payload.copy(frame, 5);
    this._sock.write(frame);
  }

  private _attachProcessGuards(): void {
    this._proc.on('exit', (code) => {
      if (this._stopped) return;
      if (!this._firstFrameSeen) this.emit('spawn-failed', new Error(`daemon exited code=${code} before first frame`));
      else this.emit('crashed', new Error(`daemon exited code=${code} mid-stream`));
    });
  }

  async stop(): Promise<void> {
    this._stopped = true;
    try { this._writeFrame(0xC9, Buffer.alloc(0)); } catch {}
    this._sock.destroy();
    if (!this._proc.killed) {
      this._proc.kill('SIGTERM');
      await new Promise((r) => setTimeout(r, 2000));
      if (!this._proc.killed) this._proc.kill('SIGKILL');
    }
  }
}
```

```typescript
import { GrpcEmuClient } from './grpc-emu-client.js';
import { scrcpyService } from './scrcpy-service.js'; // or current import path

type AndroidSession =
  | { kind: 'grpc'; client: GrpcEmuClient; display?: { width: number; height: number } }
  | { kind: 'scrcpy' };

export class AndroidStreamingService {
  private sessions = new Map<string, AndroidSession>();

  async start(device: { serial: string; kind: 'emulator' | 'physical'; grpcPort?: number | null }, ws: WebSocket | null): Promise<void> {
    const useGrpc =
      device.kind === 'emulator' &&
      device.grpcPort != null &&
      process.env.DEVICE_STREAM_ANDROID_GRPC !== '0';

    if (useGrpc) {
      try {
        const client = await GrpcEmuClient.spawn(device.serial, device.grpcPort!, ws);
        const session: AndroidSession = { kind: 'grpc', client };
        client.on('metadata', (m: { width: number; height: number }) => { session.display = m; });
        client.on('crashed', (err: Error) => log.error({ err, serial: device.serial }, 'grpc daemon crashed mid-stream'));
        this.sessions.set(device.serial, session);
        return;
      } catch (err) {
        log.warn({ err, serial: device.serial }, 'gRPC path failed, falling back to scrcpy');
      }
    }
    // Fallback (or original) scrcpy path:
    await scrcpyService.startStream(device.serial, ws);
    this.sessions.set(device.serial, { kind: 'scrcpy' });
  }

  async stop(serial: string): Promise<void> {
    const s = this.sessions.get(serial);
    if (s?.kind === 'grpc') { await s.client.stop(); }
    else { await scrcpyService.stopStream(serial); }
    this.sessions.delete(serial);
  }

  getSession(serial: string): AndroidSession | undefined { return this.sessions.get(serial); }
}
```
