/** gRPC port allocator: finds the lowest available port in band 8554-8650 (Phase 33). */
function allocateGrpcPort(usedPorts: Set<number>): number {

2. Write `server/pool/__tests__/emulator-grpc.spec.ts` (~150 LOC). Mock `node:child_process` and `node:util` `promisify(execFile)` so the driver's ADB readiness loop resolves instantly:
```typescript
import { describe, it, expect, vi, beforeEach } from 'vitest';
import { AndroidEmulatorDriver } from '../android/emulator.js';
import type { AndroidDeviceConfig } from '../types.js';

const spawnArgs: string[][] = [];
const fakeProc = { pid: 1234, unref: vi.fn(), on: vi.fn() };

vi.mock('node:child_process', () => ({
  spawn: vi.fn((_cmd, args) => { spawnArgs.push(args as string[]); return fakeProc; }),
  execFile: vi.fn((_cmd, _args, _opts, cb) => { cb?.(null, { stdout: '1', stderr: '' }); return undefined; }),
}));
vi.mock('../android/avd.js', () => ({
  ensureAvdExists: vi.fn().mockResolvedValue(undefined),
  listAvds: vi.fn().mockResolvedValue([]),
  deleteAvd: vi.fn().mockResolvedValue(undefined),
}));
vi.mock('../zombie-detector.js', () => ({
  getZombieInfo: vi.fn().mockResolvedValue({ ports: new Set(), avdNames: new Set() }),
  isProcessAlive: vi.fn().mockReturnValue(false),
  getProcessStat: vi.fn().mockResolvedValue(null),
  isZombieStat: vi.fn().mockReturnValue(false),
}));

const fakeLogger = { info: vi.fn(), warn: vi.fn(), debug: vi.fn(), error: vi.fn(), child: () => fakeLogger } as any;
const cfg: AndroidDeviceConfig = { platform: 'android', api_level: '35' /* etc */ } as AndroidDeviceConfig;

describe('AndroidEmulatorDriver -grpc spawn (AND-GRPC-SPAWN)', () => {
  beforeEach(() => { spawnArgs.length = 0; });

  it('injects -grpc <port> flag into spawn argv between -port and -no-snapshot-load', async () => {
    const driver = new AndroidEmulatorDriver(cfg, fakeLogger);
    const r = await driver.boot('test-avd', { coldBoot: true });
    expect(spawnArgs[0]).toEqual(expect.arrayContaining(['-grpc', String(r.grpcPort)]));
    const portIdx = spawnArgs[0].indexOf('-port');
    const grpcIdx = spawnArgs[0].indexOf('-grpc');
    const csIdx = spawnArgs[0].indexOf('-no-snapshot-load');
    expect(portIdx).toBeGreaterThanOrEqual(0);
    expect(grpcIdx).toBe(portIdx + 2);
    expect(csIdx).toBeGreaterThan(grpcIdx);
  });

  it('allocates distinct grpcPort for two concurrent boots (8554 then 8555)', async () => {
    const driver = new AndroidEmulatorDriver(cfg, fakeLogger);
    const r1 = await driver.boot('test-1');
    const r2 = await driver.boot('test-2');
    expect(r1.grpcPort).toBe(8554);
    expect(r2.grpcPort).toBe(8555);
  });

  it('throws when port band 8554-8650 is exhausted', async () => {
    const driver = new AndroidEmulatorDriver(cfg, fakeLogger);
    // Force-fill the internal processes Map with 97 entries holding 8554..8650
    const procs: any = (driver as any).processes;
    for (let p = 8554; p <= 8650; p++) procs.set(`busy-${p}`, { pid: p, port: 5000 + p, grpcPort: p });
    await expect(driver.boot('overflow')).rejects.toThrow(/8554-8650/);
  });

  it('returns grpcPort in BootResult', async () => {
    const driver = new AndroidEmulatorDriver(cfg, fakeLogger);
    const r = await driver.boot('test-result');
    expect(r).toHaveProperty('grpcPort');
    expect(typeof r.grpcPort).toBe('number');
    expect(r.grpcPort).toBeGreaterThanOrEqual(8554);
    expect(r.grpcPort).toBeLessThanOrEqual(8650);
  });
});
```
