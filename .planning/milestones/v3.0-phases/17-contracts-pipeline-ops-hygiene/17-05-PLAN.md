    IMPORTANT: This is a type-only smoke test. Every assertion is at compile time. The test uses Vitest's bare `test()` / `expect()` as assertion stubs, but the real work is TypeScript's typechecker accepting or rejecting the assignments. If the generated types drift (schema removed, field renamed), the test fails to compile → CI fails.

    ```typescript
    // web/src/lib/api/__tests__/generated-types.test.ts
    // Type-level smoke test — asserts that generated-types.ts exposes the expected
    // schema shapes from server/openapi.json. If the server Zod schemas change
    // and the web client has not regenerated, this file stops compiling.
    //
    // Run: npm run test (or in isolation: npx vitest run web/src/lib/api/__tests__/generated-types.test.ts)
    import { describe, test, expect } from 'vitest';
    import type { components, paths } from '../generated-types.js';

    describe('generated-types.ts — OpenAPI component schemas', () => {
      test('Hook schema exposes required fields', () => {
        // Satisfies check: if Hook's required fields drift, this fails to compile.
        const hook = {
          id: 'hook-1',
          name: 'pre-test',
          event: 'device.booted',
          command: '/usr/bin/true',
          timeout: 30000,
          env: {},
          enabled: true,
          createdAt: '2026-04-20T00:00:00.000Z',
          updatedAt: '2026-04-20T00:00:00.000Z',
        } satisfies components['schemas']['Hook'];
        expect(hook.name).toBe('pre-test');
      });

      test('JobSummary schema loads from components.schemas', () => {
        type JobSummary = components['schemas']['JobSummary'];
        // Simply assigning undefined to a typed variable exercises the type at compile time
        const j: JobSummary | undefined = undefined;
        expect(j).toBeUndefined();
      });

      test('DeviceList schema has items array', () => {
        type DeviceList = components['schemas']['DeviceList'];
        // If DeviceList drops the items field, this next line fails to compile.
        const dl: Pick<DeviceList, 'items'> | undefined = undefined;
        expect(dl).toBeUndefined();
      });

      test('paths /api/hooks supports POST', () => {
        // Compile-time check that /api/hooks has a post route
        type HooksPost = paths['/api/hooks']['post'];
        const h: HooksPost | undefined = undefined;
        expect(h).toBeUndefined();
      });
    });
    ```

    Key property: tests use `satisfies` and type-alias assignments so the real check is at `tsc`/`vitest` compile time, not at runtime. The `expect` calls are trivial — they prevent Vitest from flagging empty test bodies, not substantive assertions.
