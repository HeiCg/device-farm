       ```typescript
       import { Mutex } from 'async-mutex';
       import type pino from 'pino';
       import type { PoolEmitters } from '../../events.js';
       import type { PlatformAdapter, DiscoveredDevice } from './types.js';
       import { fingerprint, diff, sortDevices } from './fingerprint.js';

       export interface PollerDeps { adapters: PlatformAdapter[]; emit: PoolEmitters; logger: pino.Logger; intervalMs?: number; }
       export interface DeviceDiscoveryService { start(): void; stop(): void; pollOnce(): Promise<void>; getSnapshot(): DiscoveredDevice[]; }

       export function createDeviceDiscoveryService(deps: PollerDeps): DeviceDiscoveryService {
         const mutex = new Mutex();
         let prev: DiscoveredDevice[] = [];
         let lastFp = '';
         let timer: NodeJS.Timeout | null = null;
         const interval = deps.intervalMs ?? 5_000;

         async function pollOnce(): Promise<void> {
           if (mutex.isLocked()) return;
           await mutex.runExclusive(async () => {
             const lists = await Promise.all(
               deps.adapters.map((a) => a.listDevices().catch((err) => {
                 deps.logger.warn({ err }, 'discovery adapter listDevices failed');
                 return [] as DiscoveredDevice[];
               })),
             );
             const next = sortDevices(lists.flat());
             const fp = fingerprint(next);
             if (fp === lastFp) return;
             const d = diff(prev, next);
             for (const dev of d.added)   deps.emit.discoveredAdded(dev.id, { device: dev });
             for (const dev of d.removed) deps.emit.discoveredRemoved(dev.id, { device: dev });
             for (const dev of d.changed) deps.emit.discoveredChanged(dev.id, { device: dev });
             prev = next;
             lastFp = fp;
           });
         }

         function start(): void {
           if (timer) return;
           void pollOnce();
           timer = setInterval(() => { void pollOnce(); }, interval);
         }
         function stop(): void { if (timer) { clearInterval(timer); timer = null; } }
         function getSnapshot(): DiscoveredDevice[] { return prev.slice(); }
         return { start, stop, pollOnce, getSnapshot };
       }
       ```
