    ```go
    package explore

    import (
        "bytes"
        "encoding/json"
        "errors"
        "fmt"
        "io"
        "mime/multipart"
        "net/http"
        "os"
        "os/exec"
        "path/filepath"
        "regexp"
        "strings"
    )

    type StartOpts struct {
        APKPath        string
        BundleID       string
        Platform       string
        BudgetTaps     int
        BudgetScreens  int
        BudgetSeconds  int
        SeedSkeletonID string
        Model          string
    }

    type StartResponse struct {
        RunID                 string `json:"runId"`
        SessionID             string `json:"sessionId"`
        DeviceID              string `json:"deviceId"`
        AgentLogStreamURL     string `json:"agentLogStreamUrl"`
        EstimatedDurationMin  int    `json:"estimatedDurationMin"`
    }

    func StartExploration(serverURL, apiKey string, opts StartOpts) (*StartResponse, error) {
        // 1. Resolve bundle ID if omitted
        if opts.BundleID == "" {
            id, err := detectBundleID(opts.APKPath, opts.Platform)
            if err != nil {
                return nil, fmt.Errorf("--bundle-id required (auto-detection failed: %w)", err)
            }
            opts.BundleID = id
        }
        // 2. Upload APK as artifact
        artifactID, err := uploadArtifact(serverURL, apiKey, opts.APKPath, opts.Platform)
        if err != nil { return nil, fmt.Errorf("upload artifact: %w", err) }
        // 3. POST /api/explorations
        body := map[string]any{
            "appArtifactId": artifactID,
            "platform": opts.Platform,
            "bundleId": opts.BundleID,
            "budgetTaps": opts.BudgetTaps,
            "budgetScreens": opts.BudgetScreens,
            "budgetSeconds": opts.BudgetSeconds,
            "model": opts.Model,
        }
        if opts.SeedSkeletonID != "" { body["seedSkeletonId"] = opts.SeedSkeletonID }
        jsonBody, _ := json.Marshal(body)
        req, _ := http.NewRequest("POST", serverURL+"/api/explorations", bytes.NewReader(jsonBody))
        req.Header.Set("Authorization", "Bearer "+apiKey)
        req.Header.Set("Content-Type", "application/json")
        resp, err := http.DefaultClient.Do(req)
        if err != nil { return nil, fmt.Errorf("POST exploration: %w", err) }
        defer resp.Body.Close()
        if resp.StatusCode != 201 {
            body, _ := io.ReadAll(resp.Body)
            return nil, fmt.Errorf("POST /api/explorations returned %d: %s", resp.StatusCode, string(body))
        }
        var out StartResponse
        if err := json.NewDecoder(resp.Body).Decode(&out); err != nil {
            return nil, fmt.Errorf("decode response: %w", err)
        }
        return &out, nil
    }

    func uploadArtifact(serverURL, apiKey, path, platform string) (string, error) {
        // multipart/form-data — copy shape from cli/cmd/run.go uploadAPK helper.
        // POST /api/artifacts; fields: file (multipart), platform (text), type='app'.
        f, err := os.Open(path)
        if err != nil { return "", err }
        defer f.Close()
        var buf bytes.Buffer
        w := multipart.NewWriter(&buf)
        part, _ := w.CreateFormFile("file", filepath.Base(path))
        if _, err := io.Copy(part, f); err != nil { return "", err }
        _ = w.WriteField("platform", platform)
        _ = w.WriteField("type", "app")
        w.Close()
        req, _ := http.NewRequest("POST", serverURL+"/api/artifacts", &buf)
        req.Header.Set("Authorization", "Bearer "+apiKey)
        req.Header.Set("Content-Type", w.FormDataContentType())
        resp, err := http.DefaultClient.Do(req)
        if err != nil { return "", err }
        defer resp.Body.Close()
        if resp.StatusCode >= 400 {
            body, _ := io.ReadAll(resp.Body)
            return "", fmt.Errorf("upload returned %d: %s", resp.StatusCode, string(body))
        }
        var out struct{ ID string `json:"id"` }
        if err := json.NewDecoder(resp.Body).Decode(&out); err != nil { return "", err }
        return out.ID, nil
    }

    func detectBundleID(path, platform string) (string, error) {
        if platform == "android" {
            out, err := exec.Command("aapt", "dump", "badging", path).Output()
            if err != nil { return "", err }
            re := regexp.MustCompile(`package: name='([^']+)'`)
            m := re.FindStringSubmatch(string(out))
            if m == nil { return "", errors.New("could not find package name") }
            return m[1], nil
        }
        if platform == "ios" {
            // iOS .ipa contains Info.plist — unzip -p uses CFBundleIdentifier.
            cmd := exec.Command("sh", "-c", fmt.Sprintf("unzip -p '%s' '*/Info.plist' 2>/dev/null | plutil -extract CFBundleIdentifier raw - || true", path))
            out, err := cmd.Output()
            if err != nil { return "", err }
            id := strings.TrimSpace(string(out))
            if id == "" { return "", errors.New("CFBundleIdentifier not found") }
            return id, nil
        }
        return "", fmt.Errorf("unknown platform: %s", platform)
    }
    ```

    ```go
    package explore

    import (
        "encoding/json"
        "fmt"
        "net/url"
        "os"
        "strings"
        "time"

        "github.com/gorilla/websocket"
    )

    type Frame struct {
        Type          string          `json:"type"`
        TS            string          `json:"ts"`
        CorrelationID *string         `json:"correlationId"`
        Data          json.RawMessage `json:"data"`
    }

    type StreamOpts struct {
        ServerURL string
        APIKey    string
        RunID     string
        WSURL     string  // returned from StartExploration
        JSONOutput bool
        Stdout    *os.File
        Stderr    *os.File
    }

    // StreamEvents opens the WS, tails frames, prints, returns exit code per Phase 35 plan key_links.
    // Returns: 0=complete, 1=non-complete-terminal, 3=ws disconnect, 4=internal error.
    func StreamEvents(opts StreamOpts) int {
        if opts.Stdout == nil { opts.Stdout = os.Stdout }
        if opts.Stderr == nil { opts.Stderr = os.Stderr }

        u, err := url.Parse(opts.WSURL)
        if err != nil {
            fmt.Fprintf(opts.Stderr, "invalid ws url: %v\n", err)
            return 4
        }
        q := u.Query()
        q.Set("token", opts.APIKey)
        u.RawQuery = q.Encode()

        conn, _, err := websocket.DefaultDialer.Dial(u.String(), nil)
        if err != nil {
            fmt.Fprintf(opts.Stderr, "ws dial: %v\n", err)
            return 3
        }
        defer conn.Close()

        for {
            _, raw, err := conn.ReadMessage()
            if err != nil {
                fmt.Fprintf(opts.Stderr, "ws read: %v\n", err)
                return 3
            }
            var frame Frame
            if err := json.Unmarshal(raw, &frame); err != nil {
                fmt.Fprintf(opts.Stderr, "decode frame: %v\n", err)
                continue
            }
            if opts.JSONOutput {
                fmt.Fprintln(opts.Stdout, string(raw))
            } else {
                fmt.Fprintln(opts.Stdout, formatFrame(&frame))
            }
            if frame.Type == "finished" {
                var d struct{ Reason string `json:"reason"` }
                _ = json.Unmarshal(frame.Data, &d)
                if d.Reason == "complete" { return 0 }
                return 1
            }
            if frame.Type == "error" { return 1 }
        }
    }

    func formatFrame(f *Frame) string {
        ts := f.TS
        if t, err := time.Parse(time.RFC3339, f.TS); err == nil {
            ts = t.Format("15:04:05")
        }
        switch f.Type {
        case "screen-discovered":
            var d struct{ ScreenID, Title string; BFSDepth int `json:"bfsDepth"` }
            _ = json.Unmarshal(f.Data, &d)
            return fmt.Sprintf("[%s] screen-discovered: %s — %s (depth=%d)", ts, d.ScreenID, d.Title, d.BFSDepth)
        case "transition":
            var d struct{ FromScreenID, ToScreenID string; Action map[string]any `json:"action"`; IsBackEdge bool `json:"isBackEdge"` }
            _ = json.Unmarshal(f.Data, &d)
            edge := "→"
            if d.IsBackEdge { edge = "↩" }
            return fmt.Sprintf("[%s] transition: %s %s %s", ts, d.FromScreenID, edge, d.ToScreenID)
        case "tool-call":
            var d struct{ Name string; DurationMs int `json:"durationMs"` }
            _ = json.Unmarshal(f.Data, &d)
            return fmt.Sprintf("[%s] tool: %s (%dms)", ts, d.Name, d.DurationMs)
        case "stuck":
            var d struct{ ScreenID string `json:"screenId"`; Consecutive int }
            _ = json.Unmarshal(f.Data, &d)
            return fmt.Sprintf("[%s] STUCK: %s (consecutive=%d)", ts, d.ScreenID, d.Consecutive)
        case "finished":
            var d struct{ Reason string; Stats struct{ ScreensDiscovered, TransitionsTotal, TapsTaken, DurationMs int } }
            _ = json.Unmarshal(f.Data, &d)
            return fmt.Sprintf("[%s] FINISHED: %s (%d screens, %d transitions, %d taps, %dms)",
                ts, d.Reason, d.Stats.ScreensDiscovered, d.Stats.TransitionsTotal, d.Stats.TapsTaken, d.Stats.DurationMs)
        case "error":
            var d struct{ Message string }
            _ = json.Unmarshal(f.Data, &d)
            return fmt.Sprintf("[%s] ERROR: %s", ts, d.Message)
        }
        return fmt.Sprintf("[%s] %s: %s", ts, f.Type, string(f.Data))
    }
    _ = strings.HasPrefix  // keep import live if linter strips
    ```

    ```go
    import (
        "fmt"
        "os"

        "github.com/heicg/device-farm/cli/internal/config"
        "github.com/heicg/device-farm/cli/internal/explore"
        "github.com/spf13/cobra"

    var (
        exploreAPK            string
        exploreBundleID       string
        explorePlatform       string
        exploreBudgetTaps     int
        exploreBudgetScreens  int
        exploreBudgetSeconds  int
        exploreSeedSkeletonID string
        exploreModel          string
        exploreJSON           bool
    )

    var exploreCmd = &cobra.Command{
        Use:   "explore",
        Short: "Run a Claude-driven BFS exploration of an app, building a navigation graph",
        Long: `Uploads the APK/IPA, leases a device, then runs a Claude agent that
    systematically taps every screen and persists the result as an interactive graph
    viewable at /explorations/<id>.

    Examples:
      device-farm explore --apk myapp.apk --platform android
      device-farm explore --apk myapp.apk --budget-screens 80 --json
      device-farm explore --apk myapp.ipa --platform ios --seed-skeleton <id> --model claude-opus-4-7`,
        RunE: runExplore,
    }

    func init() {
        rootCmd.AddCommand(exploreCmd)
        exploreCmd.Flags().StringVar(&exploreAPK, "apk", "", "Path to APK/IPA (required)")
        exploreCmd.Flags().StringVar(&exploreBundleID, "bundle-id", "", "Bundle ID (auto-detected via aapt/Info.plist if omitted)")
        exploreCmd.Flags().StringVar(&explorePlatform, "platform", "android", "android|ios")
        exploreCmd.Flags().IntVar(&exploreBudgetTaps, "budget-taps", 200, "")
        exploreCmd.Flags().IntVar(&exploreBudgetScreens, "budget-screens", 60, "")
        exploreCmd.Flags().IntVar(&exploreBudgetSeconds, "budget-seconds", 1800, "")
        exploreCmd.Flags().StringVar(&exploreSeedSkeletonID, "seed-skeleton", "", "Optional Phase 34 iOS skeleton ID")
        exploreCmd.Flags().StringVar(&exploreModel, "model", "claude-sonnet-4-5", "Claude model name")
        exploreCmd.Flags().BoolVar(&exploreJSON, "json", false, "Emit JSON events on stdout (one per line)")
        _ = exploreCmd.MarkFlagRequired("apk")
    }

    func runExplore(cmd *cobra.Command, args []string) error {
        cfg, err := config.Load()
        if err != nil {
            os.Exit(4)
        }
        start, err := explore.StartExploration(cfg.ServerURL, cfg.APIKey, explore.StartOpts{
            APKPath:        exploreAPK,
            BundleID:       exploreBundleID,
            Platform:       explorePlatform,
            BudgetTaps:     exploreBudgetTaps,
            BudgetScreens:  exploreBudgetScreens,
            BudgetSeconds:  exploreBudgetSeconds,
            SeedSkeletonID: exploreSeedSkeletonID,
            Model:          exploreModel,
        })
        if err != nil {
            fmt.Fprintf(os.Stderr, "start exploration: %v\n", err)
            os.Exit(2)
        }
        fmt.Fprintf(os.Stderr, "Run ID: %s\nDevice: %s\nTailing events from %s ...\n",
            start.RunID, start.DeviceID, start.AgentLogStreamURL)
        code := explore.StreamEvents(explore.StreamOpts{
            ServerURL:  cfg.ServerURL,
            APIKey:     cfg.APIKey,
            RunID:      start.RunID,
            WSURL:      start.AgentLogStreamURL,
            JSONOutput: exploreJSON,
        })
        os.Exit(code)
        return nil  // unreachable
    }
    ```
