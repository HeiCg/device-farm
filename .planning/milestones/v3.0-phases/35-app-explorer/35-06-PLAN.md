    import type { Exploration, ExplorationScreen, ExplorationTransition } from '../schemas.js';

    function sidSafe(id: string): string {
      return id.replace(/[^A-Za-z0-9_]/g, '_');
    }

    function describeAction(action: Record<string, unknown>): string {
      if (action.kind === 'tap') return action.target ? `tap: ${action.target}` : 'tap';
      if (action.kind === 'swipe') return 'swipe';
      if (action.kind === 'key') return `key:${action.code}`;
      return JSON.stringify(action).slice(0, 40);
    }

    export function buildMermaid(
      screens: ExplorationScreen[],
      transitions: ExplorationTransition[],
    ): string {
      if (transitions.length === 0) return '';
      const lines = ['```mermaid', 'graph TD'];
      const titles = new Map(screens.map((s) => [s.screenId, s.title]));
      const seen = new Set<string>();
      for (const t of transitions) {
        for (const sid of [t.fromScreenId, t.toScreenId]) {
          if (!seen.has(sid)) {
            const label = titles.get(sid) ?? sid;
            lines.push(`    ${sidSafe(sid)}["${label.replace(/"/g, "'")}"]`);
            seen.add(sid);
          }
        }
      }
      for (const t of transitions) {
        const action = describeAction(t.action).replace(/"/g, "'");
        lines.push(`    ${sidSafe(t.fromScreenId)} --> |"${action}"| ${sidSafe(t.toScreenId)}`);
      }
      lines.push('```');
      return lines.join('\n');
    }

    export function enumerateJourneys(
      screens: ExplorationScreen[],
      transitions: ExplorationTransition[],
      maxPaths = 20,
    ): string[][] {
      if (screens.length === 0) return [];
      // Build adjacency from tree edges (exclude back-edges) — matches reference DFS
      const adj = new Map<string, string[]>();
      for (const t of transitions) {
        if (t.isBackEdge) continue;
        const list = adj.get(t.fromScreenId) ?? [];
        list.push(t.toScreenId);
        adj.set(t.fromScreenId, list);
      }
      const start = screens[0].screenId;
      const paths: string[][] = [];
      function dfs(node: string, path: string[]) {
        if (paths.length >= maxPaths) return;
        const newPath = [...path, node];
        const next = adj.get(node) ?? [];
        if (next.length === 0) { paths.push(newPath); return; }
        for (const n of next) {
          if (newPath.includes(n)) continue;
          dfs(n, newPath);
        }
      }
      dfs(start, []);
      return paths;
    }

    export function buildReport(
      run: Exploration,
      screens: ExplorationScreen[],
      transitions: ExplorationTransition[],
    ): string {
      const totalElements = screens.reduce((sum, s) => sum + s.elements.length, 0);
      const exploredElements = screens.reduce(
        (sum, s) => sum + s.elements.filter((e: any) => e.explored).length, 0);
      const coverage = totalElements === 0 ? 0 : Math.round((exploredElements / totalElements) * 100);
      const out: string[] = [];

      // Header
      out.push(`# Exploration Report — ${run.bundleId}`);
      out.push('');
      out.push(`**Run ID:** \`${run.id}\`  `);
      out.push(`**Platform:** ${run.platform}  `);
      out.push(`**Status:** ${run.status}  `);
      out.push(`**Started:** ${run.startedAt ? new Date(run.startedAt).toISOString() : 'n/a'}  `);
      out.push(`**Finished:** ${run.finishedAt ? new Date(run.finishedAt).toISOString() : 'n/a'}  `);
      out.push('');

      // Summary
      out.push('## Summary');
      out.push('');
      out.push('| Metric | Value |');
      out.push('|---|---|');
      out.push(`| Screens | ${screens.length} |`);
      out.push(`| Transitions | ${transitions.length} |`);
      out.push(`| Elements (total / explored) | ${totalElements} / ${exploredElements} |`);
      out.push(`| Coverage | ${coverage}% |`);
      out.push('');

      // Mermaid graph
      out.push('## Navigation Graph');
      out.push('');
      out.push(buildMermaid(screens, transitions));
      out.push('');

      // Screen inventory
      out.push('## Screen Inventory');
      out.push('');
      for (const s of screens) {
        out.push(`### ${s.title}`);
        out.push('');
        out.push(`**Screen ID:** \`${s.screenId}\` · **BFS Depth:** ${s.bfsDepth}`);
        out.push('');
        out.push(`![${s.title}](/api/artifacts/${s.screenshotArtifactId}/raw)`);
        out.push('');
        if (s.elements.length > 0) {
          out.push('**Elements:**');
          out.push('');
          for (const e of s.elements) {
            const tag = (e as any).explored ? '✓' : '○';
            const leads = (e as any).leads_to ? ` → \`${(e as any).leads_to}\`` : '';
            out.push(`- ${tag} [${(e as any).element_type}] **${(e as any).label}**${leads}`);
          }
          out.push('');
        }
        if (s.notes) { out.push(`*Notes:* ${s.notes}`); out.push(''); }
      }

      // User paths
      out.push('## User Paths');
      out.push('');
      const journeys = enumerateJourneys(screens, transitions);
      const titlesMap = new Map(screens.map((s) => [s.screenId, s.title]));
      if (journeys.length === 0) {
        out.push('_No paths found._');
      } else {
        for (let i = 0; i < journeys.length; i++) {
          const path = journeys[i];
          const labels = path.map((sid) => titlesMap.get(sid) ?? sid);
          out.push(`${i + 1}. **${labels[0]} → ${labels[labels.length-1]}** (${path.length} steps)`);
          out.push(`   ${labels.join(' → ')}`);
        }
      }
      out.push('');

      // Edge Cases (screens with notes)
      const withNotes = screens.filter((s) => s.notes && s.notes.trim().length > 0);
      if (withNotes.length > 0) {
        out.push('## Edge Cases');
        out.push('');
        out.push('| Screen | Note |');
        out.push('|---|---|');
        for (const s of withNotes) {
          out.push(`| ${s.title} | ${s.notes} |`);
        }
        out.push('');
      }

      return out.join('\n');
    }
