    ```typescript
    import type { WebSocket } from 'ws';
    import { wsExplorationFrameSchema, type WsExplorationFrame } from '../ws-schemas.js';
    import type { FastifyBaseLogger } from 'fastify';

    const RING_BUFFER_SIZE = 200;
    const HEARTBEAT_INTERVAL_MS = 30_000;

    interface RunState {
      history: WsExplorationFrame[];
      subscribers: Set<WebSocket>;
      heartbeat: NodeJS.Timeout | null;
    }

    export class ExplorationsBroadcaster {
      private runs = new Map<string, RunState>();
      constructor(private readonly logger: FastifyBaseLogger) {}

      publish(runId: string, frame: WsExplorationFrame): void {
        const parsed = wsExplorationFrameSchema.safeParse(frame);
        if (!parsed.success) {
          this.logger.warn({ runId, errors: parsed.error.issues }, 'malformed exploration ws frame dropped');
          return;
        }
        const state = this.ensureState(runId);
        state.history.push(parsed.data);
        if (state.history.length > RING_BUFFER_SIZE) state.history.shift();
        const json = JSON.stringify(parsed.data);
        for (const ws of state.subscribers) {
          if (ws.readyState === 1 /* OPEN */) {
            try { ws.send(json); } catch (err) { this.logger.warn({ err }, 'ws send failed'); }
          }
        }
      }

      subscribe(runId: string, ws: WebSocket): void {
        const state = this.ensureState(runId);
        state.subscribers.add(ws);
        // Replay history immediately:
        for (const frame of state.history) {
          try { ws.send(JSON.stringify(frame)); } catch { /* swallow */ }
        }
      }

      unsubscribe(runId: string, ws: WebSocket): void {
        const state = this.runs.get(runId);
        if (!state) return;
        state.subscribers.delete(ws);
        if (state.subscribers.size === 0 && this.isRunFinished(state)) {
          this.cleanup(runId);
        }
      }

      cleanup(runId: string): void {
        const state = this.runs.get(runId);
        if (!state) return;
        if (state.heartbeat) clearInterval(state.heartbeat);
        for (const ws of state.subscribers) { try { ws.close(); } catch {} }
        this.runs.delete(runId);
      }

      shutdown(): void {
        for (const runId of this.runs.keys()) this.cleanup(runId);
      }

      private ensureState(runId: string): RunState {
        let state = this.runs.get(runId);
        if (!state) {
          state = {
            history: [],
            subscribers: new Set(),
            heartbeat: setInterval(() => this.heartbeat(runId), HEARTBEAT_INTERVAL_MS),
          };
          this.runs.set(runId, state);
        }
        return state;
      }

      private heartbeat(runId: string): void {
        const state = this.runs.get(runId);
        if (!state) return;
        for (const ws of state.subscribers) {
          if (ws.readyState === 1) {
            try { ws.ping(); } catch {}
          }
        }
      }

      private isRunFinished(state: RunState): boolean {
        return state.history.some((f) => f.type === 'finished' || f.type === 'error');
      }
    }
    ```

    Step 2. `server/explorations/internal/events-ws.ts` (~100 lines):
    ```typescript
    import type { FastifyInstance } from 'fastify';
    import { eq } from 'drizzle-orm';
    import { explorations as explorationsTable } from '../../db/schema.js';

    export async function registerExplorationEventsWs(fastify: FastifyInstance): Promise<void> {
      fastify.get<{ Params: { id: string }; Querystring: { token?: string } }>(
        '/api/explorations/:id/events',
        { websocket: true },
        async (connection, req) => {
          const { id: runId } = req.params;
          const { token } = req.query;

          // Auth via bearer token query param (Phase 22 streaming pattern).
          if (fastify.config.auth.enabled) {
            if (!token) { connection.socket.close(4401, 'unauthorized'); return; }
            const matched = await fastify.authService.validateKey(token);
            if (!matched) { connection.socket.close(4401, 'unauthorized'); return; }
          }

          // Verify run exists.
          const [row] = await fastify.db.select().from(explorationsTable)
            .where(eq(explorationsTable.id, runId)).limit(1);
          if (!row) { connection.socket.close(4404, 'not_found'); return; }

          // Subscribe.
          fastify.explorationsModule.broadcaster.subscribe(runId, connection.socket as any);

          connection.socket.on('close', () => {
            fastify.explorationsModule.broadcaster.unsubscribe(runId, connection.socket as any);
          });
        },
      );
    }
    ```
