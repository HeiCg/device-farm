    import type { ScreenMap } from './atlas-types.js';

    export function enumeratePaths(map: ScreenMap, maxPaths = 20): string[][] {
      // DFS simple-paths enumeration from the first screen.
      // Port verbatim from _reference/.../AtlasGraph.tsx:22-40.
      if (map.screens.length === 0) return [];
      const adj = new Map<string, string[]>();
      for (const t of map.transitions) {
        if (t.is_back_edge) continue;
        const list = adj.get(t.from_screen_id) ?? [];
        list.push(t.to_screen_id);
        adj.set(t.from_screen_id, list);
      }
      const start = map.screens[0].screen_id;
      const paths: string[][] = [];
      function dfs(node: string, path: string[]) {
        if (paths.length >= maxPaths) return;
        const newPath = [...path, node];
        const next = adj.get(node) ?? [];
        if (next.length === 0) {
          paths.push(newPath);
          return;
        }
        for (const n of next) {
          if (newPath.includes(n)) continue;  // cycle-free
          dfs(n, newPath);
        }
      }
      dfs(start, []);
      return paths;
    }

    import dagre from '@dagrejs/dagre';
    import { Position, type Edge, type Node } from '@xyflow/svelte';
    import type { ScreenMap } from './atlas-types.js';

    const NODE_WIDTH = 175;
    const NODE_HEIGHT = 350;
    const MARKER_WIDTH = 160;
    const MARKER_HEIGHT = 50;

    export function layoutGraph(map: ScreenMap): { nodes: Node[]; edges: Edge[] } {
      if (!map.screens.length) return { nodes: [], edges: [] };

      // BFS depth computation + tree-edge identification — AtlasGraph.tsx:53-79.
      const adj = new Map<string, string[]>();
      for (const t of map.transitions) {
        const list = adj.get(t.from_screen_id) ?? [];
        list.push(t.to_screen_id);
        adj.set(t.from_screen_id, list);
      }
      const entry = map.screens[0].screen_id;
      const bfsDepth = new Map<string, number>([[entry, 0]]);
      const treeEdges = new Set<string>();
      const queue = [entry];
      while (queue.length) {
        const cur = queue.shift()!;
        const depth = bfsDepth.get(cur)!;
        for (const next of adj.get(cur) ?? []) {
          if (!bfsDepth.has(next)) {
            bfsDepth.set(next, depth + 1);
            treeEdges.add(`${cur}->${next}`);
            queue.push(next);
          }
        }
      }

      // Terminal screen detection — AtlasGraph.tsx:85-89.
      const terminals = map.screens
        .filter((s) => !(adj.get(s.screen_id)?.length))
        .map((s) => s.screen_id);

      // Dagre layout — AtlasGraph.tsx:95-123.
      const g = new dagre.graphlib.Graph();
      g.setGraph({ rankdir: 'TB', nodesep: 100, ranksep: 100, marginx: 60, marginy: 60 });
      g.setDefaultEdgeLabel(() => ({}));
      g.setNode('__start__', { width: MARKER_WIDTH, height: MARKER_HEIGHT });
      for (const s of map.screens) {
        g.setNode(s.screen_id, { width: NODE_WIDTH, height: NODE_HEIGHT });
      }
      for (const id of terminals) {
        g.setNode(`__end_${id}__`, { width: MARKER_WIDTH, height: MARKER_HEIGHT });
      }
      g.setEdge('__start__', entry);
      for (const t of map.transitions) {
        if (treeEdges.has(`${t.from_screen_id}->${t.to_screen_id}`)) {
          g.setEdge(t.from_screen_id, t.to_screen_id);
        }
      }
      for (const id of terminals) g.setEdge(id, `__end_${id}__`);
      dagre.layout(g);

      // Node + edge construction — AtlasGraph.tsx:127-230.
      const nodes: Node[] = [];
      const startPos = g.node('__start__');
      nodes.push({
        id: '__start__', type: 'startEnd', position: { x: startPos.x - MARKER_WIDTH/2, y: startPos.y - MARKER_HEIGHT/2 },
        data: { label: 'Start' }, sourcePosition: Position.Bottom, targetPosition: Position.Top,
      });
      for (const s of map.screens) {
        const p = g.node(s.screen_id);
        nodes.push({
          id: s.screen_id, type: 'screen',
          position: { x: p.x - NODE_WIDTH/2, y: p.y - NODE_HEIGHT/2 },
          data: { label: s.title, screenshot: s.screenshot_artifact_id, elementCount: s.elements.length, notes: s.notes },
          sourcePosition: Position.Bottom, targetPosition: Position.Top,
        });
      }
      for (const id of terminals) {
        const p = g.node(`__end_${id}__`);
        nodes.push({
          id: `__end_${id}__`, type: 'startEnd',
          position: { x: p.x - MARKER_WIDTH/2, y: p.y - MARKER_HEIGHT/2 },
          data: { label: 'End' }, sourcePosition: Position.Bottom, targetPosition: Position.Top,
        });
      }

      const edges: Edge[] = [];
      edges.push({ id: 'start->entry', source: '__start__', target: entry, type: 'smoothstep',
        style: { stroke: '#9ca3af', strokeDasharray: '6 4' } });
      for (const t of map.transitions) {
        const isTree = treeEdges.has(`${t.from_screen_id}->${t.to_screen_id}`);
        edges.push({
          id: `${t.from_screen_id}-${t.to_screen_id}-${t.bfs_order}`,
          source: t.from_screen_id, target: t.to_screen_id, type: 'smoothstep',
          label: describeAction(t.action),
          style: isTree
            ? { stroke: '#9D61FF', strokeWidth: 1.5 }
            : { stroke: '#9D61FF', strokeWidth: 1, strokeDasharray: '6 4', opacity: 0.6 },
          labelStyle: { fontFamily: 'JetBrains Mono, monospace', fontSize: 11 },
          labelBgStyle: { fill: '#1f2937', fillOpacity: 0.9 },
        });
      }
      for (const id of terminals) {
        edges.push({ id: `${id}->end`, source: id, target: `__end_${id}__`, type: 'smoothstep',
          style: { stroke: '#9ca3af', strokeDasharray: '6 4' } });
      }
      return { nodes, edges };
    }

    function describeAction(action: Record<string, unknown>): string {
      if (action.kind === 'tap') return `tap${action.target ? `: ${action.target}` : ''}`;
      if (action.kind === 'swipe') return 'swipe';
      if (action.kind === 'key') return `key:${action.code}`;
      return JSON.stringify(action).slice(0, 24);
    }

    Step 6. `ws-client.ts` (~120 lines):
    ```typescript
    import type { ScreenMap, Screen, Transition } from './atlas-types.js';

    export interface OpenWsOpts {
      runId: string;
      baseUrl: string;
      token?: string;
      onScreenDiscovered: (screen: Screen) => void;
      onTransition: (t: Transition) => void;
      onStuck: (data: { screen_id: string; consecutive: number }) => void;
      onFinished: (data: { reason: string; stats: any }) => void;
      onError: (msg: string) => void;
    }

    export function openExplorationWs(opts: OpenWsOpts): () => void {
      const url = new URL(`${opts.baseUrl.replace(/^http/, 'ws')}/api/explorations/${opts.runId}/events`);
      if (opts.token) url.searchParams.set('token', opts.token);
      const ws = new WebSocket(url.toString());
      ws.addEventListener('message', (e) => {
        try {
          const frame = JSON.parse(e.data);
          switch (frame.type) {
            case 'screen-discovered':
              opts.onScreenDiscovered({
                screen_id: frame.data.screenId, title: frame.data.title,
                screenshot_artifact_id: frame.data.screenshotArtifactId,
                elements: [], bfs_depth: frame.data.bfsDepth,
              });
              break;
            case 'transition':
              opts.onTransition({
                from_screen_id: frame.data.fromScreenId, to_screen_id: frame.data.toScreenId,
                action: frame.data.action, is_back_edge: frame.data.isBackEdge, bfs_order: 0,
              });
              break;
            case 'stuck': opts.onStuck(frame.data); break;
            case 'finished': opts.onFinished(frame.data); break;
            case 'error': opts.onError(frame.data.message); break;
          }
        } catch (err) { console.error('ws frame decode', err); }
      });
      ws.addEventListener('error', (e) => opts.onError(`ws error: ${e}`));
      return () => ws.close();
    }
    ```
