    Step 1. `server/explorations/internal/similarity.ts` (~100 lines):
    ```typescript
    import sharp from 'sharp';
    import sharpPhash from 'sharp-phash';
    import sharpPhashDistance from 'sharp-phash/distance';

    export const PHASH_THRESHOLD = 8;
    export const RMSE_THRESHOLD = 0.02;

    export async function computePhash(buf: Buffer): Promise<{ phash: string }> {
      const phash = await sharpPhash(buf);  // 64-char binary string
      return { phash };
    }

    export function hammingDistance(a: string, b: string): number {
      return sharpPhashDistance(a, b);
    }

    export async function grayscaleRmse(bufA: Buffer, bufB: Buffer): Promise<number> {
      const [pixA, pixB] = await Promise.all([
        sharp(bufA).resize(64, 64, { fit: 'fill' }).grayscale().raw().toBuffer(),
        sharp(bufB).resize(64, 64, { fit: 'fill' }).grayscale().raw().toBuffer(),
      ]);
      let sumSq = 0;
      const n = Math.min(pixA.length, pixB.length);
      for (let i = 0; i < n; i++) {
        const d = (pixA[i]! - pixB[i]!) / 255;
        sumSq += d * d;
      }
      return Math.sqrt(sumSq / n);
    }

    export interface SimilarityCandidate {
      screenId: string;
      phash: string;
      screenshotArtifactId: string;
    }

    export async function isSameScreen(
      newShot: Buffer,
      candidates: SimilarityCandidate[],
      loadArtifact: (id: string) => Promise<Buffer>,
    ): Promise<{ screenId: string; rmse: number } | null> {
      const { phash } = await computePhash(newShot);
      const close = candidates.filter((c) => hammingDistance(phash, c.phash) < PHASH_THRESHOLD);
      if (close.length === 0) return null;
      for (const c of close) {
        const candidateBuf = await loadArtifact(c.screenshotArtifactId);
        const rmse = await grayscaleRmse(newShot, candidateBuf);
        if (rmse < RMSE_THRESHOLD) return { screenId: c.screenId, rmse };
      }
      return null;
    }
    ```

    import { hammingDistance, PHASH_THRESHOLD } from './similarity.js';

    export class StuckDetector {
      private window: string[] = [];
      private readonly size: number;

      constructor(size = 3) { this.size = size; }

      observe(phash: string): { stuck: boolean; reason?: string; consecutive: number } {
        this.window.push(phash);
        if (this.window.length > this.size) this.window.shift();
        if (this.window.length < this.size) {
          return { stuck: false, consecutive: this.window.length };
        }
        const anchor = this.window[0]!;
        const allClose = this.window.every((p) => hammingDistance(p, anchor) < PHASH_THRESHOLD);
        if (allClose) {
          return { stuck: true, reason: `${this.size} consecutive taps without screen change`, consecutive: this.size };
        }
        return { stuck: false, consecutive: this.window.length };
      }

      reset(): void { this.window = []; }
    }
    ```

    export type WatchdogExitReason = 'complete' | 'budget' | 'time' | 'cancelled';

    export class Watchdog {
      private timer: NodeJS.Timeout | null;
      private exit: WatchdogExitReason | null = null;
      constructor(
        private readonly runId: string,
        private readonly budgetSeconds: number,
        private readonly onTrigger: (reason: WatchdogExitReason) => void,
      ) {
        this.timer = setTimeout(() => {
          this.exit = 'time';
          this.onTrigger('time');
        }, budgetSeconds * 1000);
      }
      shouldStop(): boolean { return this.exit !== null; }
      exitReason(): WatchdogExitReason { return this.exit ?? 'complete'; }
      tripBudget(): void {
        if (this.exit) return;
        this.exit = 'budget';
        this.onTrigger('budget');
      }
      cancel(): void {
        if (this.exit) return;
        this.exit = 'cancelled';
        this.onTrigger('cancelled');
      }
      dispose(): void {
        if (this.timer) clearTimeout(this.timer);
        this.timer = null;
      }
    }
    ```

    import { eq, and, sql } from 'drizzle-orm';
    import { createHash } from 'node:crypto';
    import type { FastifyInstance } from 'fastify';
    import {
      explorations as explorationsTable,
      explorationScreens as screensTable,
      explorationTransitions as transitionsTable,
    } from '../../db/schema.js';
    import { isSameScreen, type SimilarityCandidate } from './similarity.js';

    export function hashAction(action: Record<string, unknown>): string {
      const json = JSON.stringify(action, Object.keys(action).sort());
      return createHash('sha256').update(json).digest('hex');
    }

    export async function saveScreen(
      deps: { db: FastifyInstance['db'] },
      input: {
        runId: string;
        screenId: string;
        title: string;
        elements: any[];
        notes?: string | null;
        screenshotArtifactId: string;
        phash: string | null;
        bfsDepth: number;
      },
    ): Promise<{ screenId: string; isNew: boolean }> {
      const result = await deps.db.insert(screensTable).values({
        runId: input.runId,
        screenId: input.screenId,
        title: input.title,
        elements: input.elements,
        notes: input.notes ?? null,
        screenshotArtifactId: input.screenshotArtifactId,
        phash: input.phash,
        bfsDepth: input.bfsDepth,
      }).onConflictDoUpdate({
        target: [screensTable.runId, screensTable.screenId],
        set: {
          title: input.title,
          elements: input.elements,
          notes: input.notes ?? null,
        },
      }).returning({ id: screensTable.id });
      return { screenId: input.screenId, isNew: result.length === 1 };  // ON CONFLICT returns 1 row either way; isNew approximation
    }

    export async function saveTransition(
      deps: { db: FastifyInstance['db'] },
      input: {
        runId: string;
        fromScreenId: string;
        toScreenId: string;
        action: Record<string, unknown>;
        isBackEdge: boolean;
        bfsOrder: number;
      },
    ): Promise<{ created: boolean }> {
      const actionHash = hashAction(input.action);
      const result = await deps.db.insert(transitionsTable).values({
        runId: input.runId,
        fromScreenId: input.fromScreenId,
        toScreenId: input.toScreenId,
        action: input.action,
        actionHash,
        isBackEdge: input.isBackEdge,
        bfsOrder: input.bfsOrder,
      }).onConflictDoNothing().returning({ id: transitionsTable.id });
      return { created: result.length > 0 };
    }

    export async function getUnexplored(
      deps: { db: FastifyInstance['db'] },
      runId: string,
    ): Promise<Array<{ screenId: string; title: string; unexploredElements: any[] }>> {
      const rows = await deps.db.execute(sql`
        SELECT screen_id AS "screenId", title,
               COALESCE(jsonb_agg(elem) FILTER (WHERE NOT (elem->>'explored')::bool), '[]'::jsonb) AS "unexploredElements"
        FROM ${screensTable}, jsonb_array_elements(${screensTable.elements}) elem
        WHERE run_id = ${runId}
        GROUP BY screen_id, title
        HAVING COUNT(*) FILTER (WHERE NOT (elem->>'explored')::bool) > 0
      `);
      return rows.rows as any;
    }

    export async function getScreensForSimilarity(
      deps: { db: FastifyInstance['db'] },
      runId: string,
    ): Promise<SimilarityCandidate[]> {
      const rows = await deps.db.select({
        screenId: screensTable.screenId,
        phash: screensTable.phash,
        screenshotArtifactId: screensTable.screenshotArtifactId,
      }).from(screensTable).where(eq(screensTable.runId, runId));
      return rows.filter((r) => r.phash !== null) as SimilarityCandidate[];
    }

    export async function findSimilarScreen(
      deps: { db: FastifyInstance['db']; loadArtifact: (id: string) => Promise<Buffer> },
      runId: string,
      newShot: Buffer,
    ): Promise<{ screenId: string; rmse: number } | null> {
      const candidates = await getScreensForSimilarity(deps, runId);
      return isSameScreen(newShot, candidates, deps.loadArtifact);
    }

    export async function getScreenCount(deps: { db: FastifyInstance['db'] }, runId: string): Promise<number> {
      const [row] = await deps.db.select({ count: sql<number>`COUNT(*)::int` })
        .from(screensTable).where(eq(screensTable.runId, runId));
      return row?.count ?? 0;
    }

    export async function markRunFinished(
      deps: { db: FastifyInstance['db'] },
      runId: string,
      reason: string,
      stats: Record<string, unknown>,
      errorMessage?: string | null,
    ): Promise<void> {
      const status = reason === 'complete' ? 'complete' :
                     reason === 'budget' || reason === 'time' || reason === 'stuck' ? 'failed' :
                     reason === 'cancelled' ? 'cancelled' : 'failed';
      await deps.db.update(explorationsTable).set({
        status,
        stats,
        finishedAt: new Date(),
        errorMessage: errorMessage ?? null,
      }).where(eq(explorationsTable.id, runId));
    }

    export async function markRunStarted(deps: { db: FastifyInstance['db'] }, runId: string): Promise<void> {
      await deps.db.update(explorationsTable).set({
        status: 'running',
        startedAt: new Date(),
      }).where(eq(explorationsTable.id, runId));
    }

    import { tool } from '@anthropic-ai/claude-agent-sdk';
    import { z } from 'zod';
    import * as store from './store.js';
    import { computePhash } from './similarity.js';
    import { StuckDetector } from './stuck-detector.js';
    import type { FastifyInstance } from 'fastify';
    import type { ExplorationsModule } from './module.js';

    export interface AgentToolDeps {
      fastify: FastifyInstance;
      explorationsModule: ExplorationsModule;
      stuckDetector: StuckDetector;
      tapCounter: { value: number; budgetTaps: number };
      budgetScreens: number;
      loadArtifact: (id: string) => Promise<Buffer>;
      bfsOrderCounter: { value: number };
    }

    export function buildExplorationTools(runId: string, deps: AgentToolDeps) {
      return [
        tool('explore_save_screen', 'Save a newly-discovered screen ...',
          {
            title: z.string().min(1).max(200),
            elements: z.array(z.object({
              label: z.string(), element_type: z.enum(['button','input','link','tab','list_item','icon','text']),
              notes: z.string().optional(),
            })),
            notes: z.string().optional(),
            screenshot_artifact_id: z.string().uuid(),
          },
          async ({ title, elements, notes, screenshot_artifact_id }) => {
            const screenCount = await store.getScreenCount({ db: deps.fastify.db }, runId);
            if (screenCount >= deps.budgetScreens) {
              return {
                content: [{ type: 'text', text: `Budget exceeded: ${screenCount}/${deps.budgetScreens} screens. Call explore_finish.` }],
                structuredContent: { error: 'budget_exceeded', screenCount },
              };
            }
            const buf = await deps.loadArtifact(screenshot_artifact_id);
            const { phash } = await computePhash(buf);
            const existing = await store.findSimilarScreen(
              { db: deps.fastify.db, loadArtifact: deps.loadArtifact }, runId, buf,
            );
            const stuck = deps.stuckDetector.observe(phash);
            if (existing) {
              if (stuck.stuck) {
                await deps.explorationsModule.emit.stuck({ runId, screenId: existing.screenId, consecutive: stuck.consecutive });
              }
              return {
                content: [{ type: 'text', text: `Duplicate of ${existing.screenId} (rmse=${existing.rmse.toFixed(4)})${stuck.stuck ? `; STUCK consecutive=${stuck.consecutive}` : ''}` }],
                structuredContent: { screen_id: existing.screenId, isDuplicate: true, matched: existing.screenId, stuckCount: stuck.consecutive },
              };
            }
            const screenId = title.toLowerCase().replace(/[^a-z0-9]+/g, '-').slice(0, 40);
            await store.saveScreen({ db: deps.fastify.db }, {
              runId, screenId, title, elements: elements.map((e) => ({ ...e, explored: false })),
              notes, screenshotArtifactId: screenshot_artifact_id, phash, bfsDepth: 0,
            });
            await deps.explorationsModule.emit.screenDiscovered({
              runId, screenId, title, bfsDepth: 0, screenshotArtifactId: screenshot_artifact_id,
            });
            return {
              content: [{ type: 'text', text: `Saved screen "${title}" as ${screenId}` }],
              structuredContent: { screen_id: screenId, isDuplicate: false },
            };
          },
        ),
        tool('explore_save_transition', '...',
          {
            from_screen_id: z.string(), to_screen_id: z.string(),
            action: z.record(z.string(), z.unknown()),
            is_back_edge: z.boolean().default(false),
          },
          async ({ from_screen_id, to_screen_id, action, is_back_edge }) => {
            const order = deps.bfsOrderCounter.value++;
            const { created } = await store.saveTransition({ db: deps.fastify.db }, {
              runId, fromScreenId: from_screen_id, toScreenId: to_screen_id,
              action, isBackEdge: is_back_edge, bfsOrder: order,
            });
            if (created) {
              await deps.explorationsModule.emit.transition({
                runId, fromScreenId: from_screen_id, toScreenId: to_screen_id, action, isBackEdge: is_back_edge,
              });
            }
            return {
              content: [{ type: 'text', text: created ? 'Transition recorded' : 'Transition already exists (skipped)' }],
              structuredContent: { ok: true, created },
            };
          },
        ),
        tool('explore_mark_element_explored', '...', {
          screen_id: z.string(), element_label: z.string(), leads_to: z.string().nullable().optional(),
        }, async ({ screen_id, element_label, leads_to }) => {
          // UPDATE elements jsonb — set element.explored=true where label=element_label
          await deps.fastify.db.execute(sql`
            UPDATE exploration_screens
            SET elements = (
              SELECT jsonb_agg(
                CASE WHEN elem->>'label' = ${element_label}
                  THEN jsonb_set(jsonb_set(elem, '{explored}', 'true'::jsonb),
                                 '{leads_to}', ${leads_to ? `"${leads_to}"` : 'null'}::jsonb)
                  ELSE elem
                END
              ) FROM jsonb_array_elements(elements) elem
            )
            WHERE run_id = ${runId} AND screen_id = ${screen_id}
          `);
          return { content: [{ type: 'text', text: 'Marked' }], structuredContent: { ok: true } };
        }),
        tool('explore_get_unexplored', 'Return screens with unexplored elements', {},
          async () => {
            const rows = await store.getUnexplored({ db: deps.fastify.db }, runId);
            return {
              content: [{ type: 'text', text: JSON.stringify(rows) }],
              structuredContent: { screens: rows },
            };
          },
        ),
        tool('explore_finish', 'Mark the exploration complete', {
          reason: z.enum(['complete', 'budget', 'stuck', 'login_required']),
          message: z.string().optional(),
        }, async ({ reason, message }) => {
          // Mark intent — the agent-runner finalizes after the query returns.
          return {
            content: [{ type: 'text', text: `Finish requested: ${reason}${message ? ` — ${message}` : ''}` }],
            structuredContent: { finishRequested: true, reason, message },
          };
        }),
      ];
    }

    import type PgBoss from 'pg-boss';
    import { z } from 'zod';
    import { runExploration, type ExplorationRunnerDeps } from './internal/agent-runner.js';

    export const EXPLORATION_RUN_QUEUE_NAME = 'exploration.run' as const;

    export const explorationRunPayloadSchema = z.object({
      runId: z.string().uuid(),
    });
    export type ExplorationRunPayload = z.infer<typeof explorationRunPayloadSchema>;

    export async function registerExplorationsRunWorker(
      boss: PgBoss,
      deps: ExplorationRunnerDeps,
    ): Promise<string> {
      await boss.createQueue(EXPLORATION_RUN_QUEUE_NAME, {
        name: EXPLORATION_RUN_QUEUE_NAME,
        policy: 'stately',
        retryLimit: 0,  // agent runs are not retryable — partial state preserved
        expireInSeconds: 7200,  // 2h cap (matches max budgetSeconds)
      } as any);
      const workerId = await boss.work<ExplorationRunPayload>(
        EXPLORATION_RUN_QUEUE_NAME,
        { batchSize: 1 },
        async ([job]) => {
          const payload = explorationRunPayloadSchema.parse(job.data);
          await runExploration(payload.runId, deps);
        },
      );
      return workerId;
    }
